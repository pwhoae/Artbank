# 1. Player.gd

對應 HTML 的：

```js
class Player
```

Godot：

```gdscript
extends CharacterBody2D

@export var speed := 250.0

var hp := 100
var max_hp := 100

var laser_lv := 1
var shield_lv := 0
var grenade_lv := 0

func _physics_process(delta):

    var input_dir := Input.get_vector(
        "move_left",
        "move_right",
        "move_up",
        "move_down"
    )

    velocity = input_dir * speed
    move_and_slide()

    if input_dir != Vector2.ZERO:
        rotation = input_dir.angle()
```

```
WASD
Arrow Keys
```

# 2. Enemy.gd

對應：

```js
class Enemy
```

Godot：

```gdscript
extends CharacterBody2D

enum EnemyType {
    HOUND,
    ENFORCER,
    BOSS
}

@export var enemy_type : EnemyType

var hp
var damage
var speed

func _ready():

    match enemy_type:

        EnemyType.HOUND:
            hp = 22
            damage = 8
            speed = 120

        EnemyType.ENFORCER:
            hp = 80
            damage = 18
            speed = 70

        EnemyType.BOSS:
            hp = 1000
            damage = 25
            speed = 50


func update_ai(player_pos):

    var dir = (player_pos - global_position).normalized()

    velocity = dir * speed
    move_and_slide()


func hit(amount):

    hp -= amount

    if hp <= 0:
        die()


func die():
    queue_free()
```

# 3. EnemyManager.gd

對應：

```js
spawnEnemyRandomly()
```

Godot：

```gdscript
extends Node

@export var enemy_scene : PackedScene

var spawn_timer := 0.0

func _process(delta):

    spawn_timer += delta

    if spawn_timer >= 1.0:

        spawn_timer = 0

        spawn_enemy()


func spawn_enemy():

    var enemy = enemy_scene.instantiate()

    var player = get_tree().get_first_node_in_group("player")

    var angle = randf()*TAU
    var dist = 800

    enemy.global_position =
        player.global_position +
        Vector2.RIGHT.rotated(angle)*dist

    get_parent().add_child(enemy)
```

# 4. Laser Weapon

HTML：

```js
LaserBeam
```

Godot 建議：

不用生成實體子彈。

直接：

```gdscript
func fire_laser():

    var target = find_nearest_enemy()

    if target == null:
        return

    target.hit(18 + player.laser_lv * 4)

    spawn_laser_effect(
        player.global_position,
        target.global_position
    )
```

---

Laser 特效：

```gdscript
extends Line2D

var timer := 0.08

func setup(start,end):

    points = [start,end]

func _process(delta):

    timer -= delta

    if timer <= 0:
        queue_free()
```

# 5. Shield Orbit

HTML：

```js
ShieldOrbit
```

Godot：

```gdscript
extends Node2D

var angle := 0.0
var level := 1

func _process(delta):

    angle += delta*4

    for i in get_children().size():

        var orb = get_child(i)

        var a = angle + TAU*i/level

        orb.position =
            Vector2(cos(a),sin(a))
            *(55+level*8)
```

# 7. Exp Gem

HTML：

```js
ExpGem
```

Godot：

```gdscript
extends Area2D

var exp_value := 15

func _process(delta):

    var player = get_tree().get_first_node_in_group("player")

    var dist = global_position.distance_to(player.global_position)

    if dist < 150:

        global_position += (
            player.global_position-global_position
        ).normalized()*300*delta


func _on_body_entered(body):

    if body.is_in_group("player"):

        body.add_exp(exp_value)

        queue_free()
```

# 8. Level Up 系統

HTML：

```js
triggerLevelUp()
```

Godot：

Player：

```gdscript
var exp := 0
var level := 1
var next_exp := 60

func add_exp(value):

    exp += value

    if exp >= next_exp:

        level_up()


func level_up():

    level += 1

    exp -= next_exp

    next_exp *= 1.35

    get_tree().paused = true

    UI.show_upgrade()
```

---

Upgrade UI：

```gdscript
func choose_laser():

    player.laser_lv += 1

    resume()


func choose_shield():

    player.shield_lv += 1

    resume()


func choose_grenade():

    player.grenade_lv += 1

    resume()


func resume():

    visible = false

    get_tree().paused = false
```

# 10. Main Game Loop

HTML：

```js
requestAnimationFrame(gameLoop)
```

Godot 不需要。

Godot 自帶遊戲循環：

```gdscript
func _process(delta):
    update_ui()

func _physics_process(delta):
    player_logic()
    enemy_logic()
```
