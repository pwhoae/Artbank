推薦的 Godot 4 節點結構 (Scene Tree)

在 Godot 4 中，為了實現完美的拖拽、縮放與遮擋關係，我們建議使用 Node2D 而非單純的 Control UI。這可以避開 UI 容器（如 HBoxContainer）強行限制子節點 Position 和 Rotation 的問題。

📂 MainScene (Node2D) # 遊戲主場景
├── 📂 SlotsContainer (Node2D) # 放置插槽的容器
├── 📂 Slot1 (Area2D) # 帶有 CollisionShape2D 作為吸附感應區
├── 📂 Slot2 (Area2D)
└── 📂 Slot3 (Area2D)
├── 📂 HandContainer (Node2D) # 核心：手牌管理器，掛載 HandManager.gd
├── 📄 Card1 (Area2D 或 Control) # 動態生成的卡牌實例
└── 📄 Card2 (Area2D 或 Control)
└── 📂 PlayerCharacter (CharacterBody2D) # 中間的小黑人

卡牌節點 (Card.tscn)

  • • 根節點： Area2D (方便進行滑鼠點擊與拖拽碰撞檢測)
  • • 外觀： Sprite2D 或 TextureRect 顯示手繪風卡牌圖案
  • • 文字： Label 顯示數值
  • • 遮罩： 設置 z_index 用於控制懸停時置頂

插槽節點 (Slot.tscn)

  • • 根節點： Area2D
  • • 碰撞體： CollisionShape2D (矩形，半徑約卡牌大小)
  • • 背景： Sprite2D 繪製虛線或半透明外框
  • • 狀態變數： 儲存 is_occupied 及佔用它的卡牌引用

# 核心演算法：計算並平滑插值排列所有手牌
func update_hand_layout():
    var card_count = cards.size()
    if card_count == 0: return
    
    for i in range(card_count):
        var card = cards[i]
        if card.is_dragging: continue # 被拖拽的卡牌不參與自動排列
        
        # 1. 計算比率 t (-1.0 到 1.0)
        var t = 0.0
        if card_count > 1:
            t = (float(i) / (card_count - 1)) * 2.0 - 1.0
            
        # 2. 計算 X 座標偏置
        var target_x = t * (hand_width / 2.0)
        
        # 3. 計算 Y 座標偏置（二次曲線實現圓弧效果）
        # 當 t = 0 (中間) 時，拱起最大；t = -1 或 1 時，高度降至最低
        var target_y = (1.0 - (t * t)) * -height_curve_offset
        
        # 4. 計算旋轉角度
        var target_rot = t * deg_to_rad(max_rotation_degrees)
        
        # 5. 使用 Tween 進行平滑移位
        var tween = create_tween().set_parallel(true)
        tween.tween_property(card, "position", Vector2(target_x, target_y), 0.25)\
            .set_trans(Tween.TRANS_SINE).set_ease(Tween.EASE_OUT)
        tween.tween_property(card, "rotation", target_rot, 0.25)\
            .set_trans(Tween.TRANS_SINE).set_ease(Tween.EASE_OUT)
            
        # 確保遮擋關係正確：從左到右 Z-Index 遞增
        card.z_index = i

# Card.gd (掛載於卡牌單體)
extends Area2D

signal drag_started
signal drag_ended(card, final_position)

var is_dragging: bool = false
var drag_offset: Vector2 = Vector2.ZERO
var original_z_index: int = 0

# 吸附檢測相關參數
var snap_target_slot: Area2D = null
var snap_distance_threshold: float = 60.0

func _ready():
    # 連結滑鼠輸入信號
    input_event.connect(_on_input_event)

func _process(delta):
    if is_dragging:
        var mouse_pos = get_global_mouse_position()
        var target_pos = mouse_pos - drag_offset
        
        # 核心：即時尋找最近的插槽並進行磁性吸附預覽
        var nearest_slot = find_nearest_empty_slot(target_pos)
        
        if nearest_slot:
            # 磁性吸附：如果距離小於閾值，卡牌平滑吸附在插槽位置，並呈現預覽狀態
            global_position = global_position.lerp(nearest_slot.global_position, 0.25)
            rotation = lerp_angle(rotation, 0, 0.25) # 吸附時轉正
            scale = scale.lerp(Vector2(0.9, 0.9), 0.2) # 稍微縮小表示合體
            snap_target_slot = nearest_slot
        else:
            # 一般自由拖拽跟隨滑鼠
            global_position = target_pos
            rotation = lerp_angle(rotation, 0, 0.1)
            scale = scale.lerp(Vector2(1.1, 1.1), 0.2) # 放大表示正在被提起
            snap_target_slot = null

func _on_input_event(viewport, event, shape_idx):
    if event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT:
        if event.pressed:
            start_drag()
        elif !event.pressed and is_dragging:
            end_drag()

func start_drag():
    is_dragging = true
    drag_offset = get_global_mouse_position() - global_position
    original_z_index = z_index
    z_index = 100 # 置頂顯示
    drag_started.emit()

func end_drag():
    is_dragging = false
    z_index = original_z_index
    
    if snap_target_slot:
        # 正式放入插槽
        global_position = snap_target_slot.global_position
        snap_target_slot.occupy(self)
        drag_ended.emit(self, true) # 成功落袋
    else:
        # 回歸手牌排列
        drag_ended.emit(self, false) # 需要重置回原位

func find_nearest_empty_slot(pos: Vector2) -> Area2D:
    var slots = get_tree().get_nodes_in_group("slots")
    var nearest: Area2D = null
    var min_dist = snap_distance_threshold
    
    for slot in slots:
        if slot.is_occupied: continue
        var dist = pos.distance_to(slot.global_position)
        if dist < min_dist:
            min_dist = dist
            nearest = slot
    return nearest

還原 Unity 影片效果的 3 大關鍵技術

1. 扇形排列 (Fan Hand Layout)

影片中，手牌不是呈一條直線，而是像一把扇子。這是利用數學上的二次曲線（Quadratic Bezier 或簡單的拋物線）計算出來的。在程式碼中，我們將手牌的排列係數 $t$ 限制在 $[-1.0, 1.0]$。 Y 軸高度偏置公式為： $Height(t) = (1 - t^2) \times Offset$。這能確保正中間的卡牌高度最高，而兩端漸漸垂下，配合旋轉，形成完美弧形。

2. 懸停置頂與縮放 (Z-Index Hover)

當滑鼠移入時，該卡牌必須被放到最上層，否則會被隔壁卡牌遮擋。在 Godot 中，我們只需要在檢測到滑鼠 Hover 時，臨時將卡牌的 z_index 設為很高（例如 99），並利用 Tween 對其 scale、position.y 進行微調（向上浮動 20-30 像素），就能獲得極佳的打牌手感。

3. 磁性吸附預覽 (Magnetic Snapping Preset)

影片最亮眼的是拖動卡牌靠近插槽時的**吸附效果**。當玩家釋放滑鼠前，卡牌會提前「吸附」在槽位中，給予強烈的視覺反饋。 我們在每一幀物理幀 (_process) 中，即時搜尋距離卡牌最近且未被佔用的插槽。若距離小於閾值，我們使用插值 lerp 讓卡牌向插槽座標靠攏，放開滑鼠時則將其完全繫定到插槽上。
