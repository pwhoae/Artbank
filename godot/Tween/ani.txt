class_name CharacterRevealIntro
extends Node2D

Godot 4 Character Intro Animation Controller

Recreates the stylized anime character reveal sequence (whip pans, zoom springs, shader blurs, UI slide-ins).

var _shake_decay: float = 5.0
var _shader_material: ShaderMaterial

Embedded Godot 4 CanvasItem Shader Code for Screen Effects

const SHADER_CODE: String = """
shader_type canvas_item;

uniform sampler2D SCREEN_TEXTURE : hint_screen_texture, filter_linear_mipmap;
uniform float blur_amount : hint_range(0.0, 0.05) = 0.0;
uniform float chromatic_aberration : hint_range(0.0, 0.03) = 0.0;
uniform float radial_blur_strength : hint_range(0.0, 0.1) = 0.0;
uniform vec2 focus_center = vec2(0.5, 0.5);

void fragment() {
vec2 uv = SCREEN_UV;
vec2 dir = uv - focus_center;

// Chromatic Aberration + Radial Blur Sampling
vec4 color = vec4(0.0);
int samples = 8;

for (int i = 0; i < samples; i++) {
	float scale = 1.0 - radial_blur_strength * (float(i) / float(samples));
	vec2 sampled_uv = (uv - focus_center) * scale + focus_center;
	
	// Offset R and B channels for chromatic aberration
	vec2 ca_offset = dir * chromatic_aberration;
	float r = texture(SCREEN_TEXTURE, sampled_uv - ca_offset).r;
	float g = texture(SCREEN_TEXTURE, sampled_uv).g;
	float b = texture(SCREEN_TEXTURE, sampled_uv + ca_offset).b;
	float a = texture(SCREEN_TEXTURE, sampled_uv).a;
	
	color += vec4(r, g, b, a);
}

COLOR = color / float(samples);


}
"""

STREAMING_CHUNK:Configuring lifecycle callbacks...

func _ready() -> void:
_setup_shader_material()
_reset_initial_state()
# Automatically play intro on start
play_reveal_sequence()

func _process(delta: float) -> void:
_process_camera_shake(delta)

STREAMING_CHUNK:Building shader initialization logic...

func _setup_shader_material() -> void:
var shader = Shader.new()
shader.code = SHADER_CODE
_shader_material = ShaderMaterial.new()
_shader_material.shader = shader

if blur_color_rect:
	blur_color_rect.material = _shader_material
	blur_color_rect.visible = true


STREAMING_CHUNK:Setting up initial scene state...

func _reset_initial_state() -> void:
if camera:
camera.position = face_position + Vector2(0, -100)
camera.zoom = Vector2(2.8, 2.8)
camera.rotation = deg_to_rad(-2.0)

if ui_layer:
	if background_text_label:
		background_text_label.modulate.a = 0.0
		background_text_label.scale = Vector2(1.2, 1.2)
	if character_card:
		character_card.modulate.a = 0.0
		character_card.position.x += 100.0
	if vinyl_record:
		vinyl_record.position.y += 300.0

if transition_arrow:
	transition_arrow.visible = false
	transition_arrow.scale = Vector2.ZERO


STREAMING_CHUNK:Orchestrating main sequence tween...

func play_reveal_sequence() -> void:
if _is_animating:
return
_is_animating = true

# Kill any existing tween
if _tween and _tween.is_valid():
	_tween.kill()

_tween = create_tween().set_parallel(true)

# -------------------------------------------------------------
# FRAME 00:00 - 00:02 | SHOT 1: Extreme Close-Up on Head / Face
# Fast snap to face with intense blur & chromatic aberration decay
# -------------------------------------------------------------
_tween.chain().tween_property(camera, "position", face_position, 0.35) \
	.set_trans(Tween.TRANS_EXPO).set_ease(Tween.EASE_OUT)
_tween.tween_property(camera, "zoom", Vector2(2.2, 2.2), 0.35) \
	.set_trans(Tween.TRANS_EXPO).set_ease(Tween.EASE_OUT)
_tween.tween_property(camera, "rotation", deg_to_rad(0.0), 0.35) \
	.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)

# Shader parameters during whip snap
_tween.tween_method(_set_radial_blur, 0.08, 0.0, 0.4) \
	.set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
_tween.tween_method(_set_chromatic_aberration, 0.02, 0.001, 0.4)

# Camera Shake on impact
_tween.chain().tween_callback(func(): _trigger_shake(8.0))

# Hold on face briefly
_tween.chain().tween_interval(0.25)

# -------------------------------------------------------------
# FRAME 00:03 - 00:04 | SHOT 2: Transition Arrow & Whip Pan to Boots
# Sudden blue arrow overlay snap, fast downward zoom pan
# -------------------------------------------------------------
_tween.chain().tween_callback(_play_transition_arrow_anim)

_tween.tween_property(camera, "position", boots_position, 0.3) \
	.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_IN_OUT)
_tween.tween_property(camera, "zoom", Vector2(1.9, 1.9), 0.3) \
	.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_IN_OUT)

_tween.tween_method(_set_radial_blur, 0.06, 0.0, 0.3)

# Hold on boots/legs
_tween.chain().tween_interval(0.2)

# -------------------------------------------------------------
# FRAME 00:05 - 00:07 | SHOT 3: Dynamic Pull-back to Full Body
# Zoom out with spring/elastic overshoot (EaseOutBack)
# -------------------------------------------------------------
_tween.chain().tween_property(camera, "position", full_body_position, 0.55) \
	.set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)
_tween.tween_property(camera, "zoom", Vector2(1.0, 1.0), 0.55) \
	.set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)

_tween.tween_method(_set_radial_blur, 0.04, 0.0, 0.3)
_tween.tween_callback(func(): _trigger_shake(12.0))

# -------------------------------------------------------------
# FRAME 00:08 - 00:14 | SHOT 4: UI Graphic Cards & Text Pop-in
# "NORMA" typography background, profile card slide-in, vinyl record
# -------------------------------------------------------------
_tween.chain().tween_callback(_animate_ui_elements)

# Infinite subtle camera float / breathing effect after sequence ends
_tween.chain().tween_callback(_start_idle_camera_float)


STREAMING_CHUNK:Animating UI element arrivals...

func _animate_ui_elements() -> void:
var ui_tween = create_tween().set_parallel(true)

# Background "NORMA" Typography scale down & fade in
if background_text_label:
	ui_tween.tween_property(background_text_label, "modulate:a", 1.0, 0.35)
	ui_tween.tween_property(background_text_label, "scale", Vector2.ONE, 0.45) \
		.set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)

# Character Details Card slide-in from right
if character_card:
	ui_tween.tween_property(character_card, "modulate:a", 1.0, 0.3)
	ui_tween.tween_property(character_card, "position:x", character_card.position.x - 100.0, 0.4) \
		.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)

# Bottom Vinyl Record slide up
if vinyl_record:
	ui_tween.tween_property(vinyl_record, "position:y", vinyl_record.position.y - 300.0, 0.5) \
		.set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)


STREAMING_CHUNK:Executing transition arrow overlay effects...

func _play_transition_arrow_anim() -> void:
if not transition_arrow:
return
transition_arrow.visible = true
transition_arrow.scale = Vector2(0.2, 0.2)
transition_arrow.modulate.a = 1.0

var arrow_tween = create_tween().set_parallel(true)
arrow_tween.tween_property(transition_arrow, "scale", Vector2(1.2, 1.2), 0.25) \
	.set_trans(Tween.TRANS_EXPO).set_ease(Tween.EASE_OUT)
arrow_tween.chain().tween_property(transition_arrow, "modulate:a", 0.0, 0.15)


STREAMING_CHUNK:Handling camera shake physics...

func _trigger_shake(amount: float) -> void:
_shake_intensity = amount

func _process_camera_shake(delta: float) -> void:
if _shake_intensity > 0.0 and camera:
camera.offset = Vector2(
randf_range(-_shake_intensity, _shake_intensity),
randf_range(-_shake_intensity, _shake_intensity)
)
_shake_intensity = move_toward(_shake_intensity, 0.0, _shake_decay * delta * 10.0)
elif camera and camera.offset != Vector2.ZERO:
camera.offset = Vector2.ZERO

STREAMING_CHUNK:Implementing idle floating animation...

func _start_idle_camera_float() -> void:
_is_animating = false
var float_tween = create_tween().set_loops().set_parallel(true)
float_tween.tween_property(camera, "position:y", full_body_position.y - 6.0, 2.5) 

.set_trans(Tween.TRANS_SINE).set_ease(Tween.EASE_IN_OUT)
float_tween.chain().tween_property(camera, "position:y", full_body_position.y + 6.0, 2.5) 

.set_trans(Tween.TRANS_SINE).set_ease(Tween.EASE_IN_OUT)

STREAMING_CHUNK:Updating shader uniform parameters...

func _set_radial_blur(val: float) -> void:
if _shader_material:
_shader_material.set_shader_parameter("radial_blur_strength", val)

func _set_chromatic_aberration(val: float) -> void:
if _shader_material:
_shader_material.set_shader_parameter("chromatic_aberration", val)
