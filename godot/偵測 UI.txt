Node2D + Sprite + outline + UI跟隨

timer-based progress

# ⚙️ 三、完整 Godot 4 Code（可直接用）

## 🎯 1. Scanner.gd（核心邏輯）

```gdscript
extends Node2D

@export var scan_time := 2.0
@export var scan_radius := 200.0

var current_target = null
var scan_progress := 0.0

@onready var area = $ScannerArea
@onready var ui = $ScannerUI

func _ready():
    area.body_entered.connect(_on_body_entered)
    area.body_exited.connect(_on_body_exited)

func _process(delta):
    if current_target:
        scan_progress += delta / scan_time
        scan_progress = clamp(scan_progress, 0, 1)

        ui.update_progress(scan_progress)
        ui.update_target_position(current_target.global_position)

        if scan_progress >= 1:
            complete_scan()
    else:
        # 沒目標 → 慢慢下降
        scan_progress = max(scan_progress - delta * 0.5, 0)
        ui.update_progress(scan_progress)

func _on_body_entered(body):
    if body.is_in_group("scannable"):
        current_target = body
        ui.show_target(body)

func _on_body_exited(body):
    if body == current_target:
        current_target = null
        ui.hide_target()

func complete_scan():
    print("掃描完成:", current_target.name)
    ui.show_info(current_target)
```

## 🎯 2. ScannerUI.gd（UI控制）

```gdscript
extends CanvasLayer

@onready var progress = $ScanCircle
@onready var target_icon = $TargetIcon
@onready var info_panel = $InfoPanel

func update_progress(value):
    progress.value = value * 100

func show_target(target):
    target_icon.visible = true

func hide_target():
    target_icon.visible = false
    info_panel.visible = false

func update_target_position(world_pos):
    var cam = get_viewport().get_camera_2d()
    if cam:
        var screen_pos = cam.unproject_position(world_pos)
        target_icon.position = screen_pos

func show_info(target):
    info_panel.visible = true
    info_panel.get_node("Name").text = "名稱: " + target.name
    info_panel.get_node("Status").text = "狀態: 已掃描"
```

# 🎨 四、掃描圓 UI 做法（重點）

## 方法 A（最簡單）

👉 用 `TextureProgressBar`

設定：

```
Fill Mode → Radial
Min = 0
Max = 100
```

---

## 方法 B（高級：Shader）

```glsl
shader_type canvas_item;

uniform float progress : hint_range(0.0, 1.0);

void fragment() {
    vec2 uv = UV - vec2(0.5);
    float angle = atan(uv.y, uv.x);

    float normalized = (angle + PI) / (2.0 * PI);

    if (normalized < progress) {
        COLOR = vec4(0.0, 1.0, 1.0, 1.0);
    } else {
        discard;
    }
}
```
