高级三维空间拖尾着色器 (Spatial Shader)

支持噪声扭曲、UV流动与渐变贴图（完全契合 image_df16cb.png 底图设计）

shader_type spatial;
render_mode blend_add, depth_draw_opaque, cull_disabled, unshaded;

/* 
 * Godot 4 拖尾纹理扭曲着色器 (VFX Trail Shader)
 * 完美还原 image_df16cb.png 的多重流动、高光晕边缘与溶解效果
 */

uniform sampler2D main_texture : source_color, filter_linear_mipmap, repeat_enable;
uniform sampler2D noise_texture : filter_linear_mipmap, repeat_enable;

group_uniforms Trail_Styling;
uniform vec4 trail_color : source_color = vec4(0.55, 0.15, 0.95, 1.0); // 核心紫色
uniform vec4 glow_color : source_color = vec4(0.9, 0.4, 1.0, 1.0);     // 边缘发光色
uniform float emission_intensity : hint_range(1.0, 8.0) = 4.0;          // 发光强度

group_uniforms Motion;
uniform float scroll_speed = 2.0;               // 纹理沿拖尾流动的速度
uniform float distortion_strength = 0.18;       // 噪声扭曲强度 (像火焰/风一样肆虐)
uniform float distortion_speed = 1.5;           // 扭曲扰动频率

group_uniforms Mask;
uniform float fade_power : hint_range(0.5, 4.0) = 1.5; // 控制首尾自然淡入淡出的指数

void vertex() {
    // 拖尾网格在动态生成时会传入顶点色
    // COLOR.a 一般代表拖尾的生存寿命衰减 (0 = 消失, 1 = 刚生成)
}

void fragment() {
    // 1. 噪声扭曲计算
    vec2 noise_uv = UV * 1.5 + vec2(-TIME * distortion_speed * 0.5, TIME * distortion_speed * 0.2);
    vec4 noise_val = texture(noise_texture, noise_uv);
    
    // 使用噪声的红绿通道作为UV偏移向量，让流动产生弯曲和破裂感
    vec2 distorted_uv = UV;
    distorted_uv.x -= TIME * scroll_speed;
    distorted_uv += (noise_val.rg - vec2(0.5)) * distortion_strength;

    // 2. 采样基础拖尾纹理 (即 image_df16cb.png 底部展示的黑白灰度遮罩)
    vec4 tex_color = texture(main_texture, distorted_uv);
    
    // 3. 计算拖尾边缘羽化与渐变 (UV.x 控制长度, UV.y 控制宽度)
    // 侧边羽化，使拖尾在边缘两端自然渐隐
    float edge_fade = smoothstep(0.0, 0.15, UV.y) * smoothstep(1.0, 0.85, UV.y);
    // 尾部寿命衰变，配合顶点颜色Alpha
    float life_fade = pow(UV.x, fade_power) * COLOR.a;
    
    float final_alpha = tex_color.r * edge_fade * life_fade;
    
    // 4. 色彩映射与发光
    // 融合主色调与高亮发光色，高浓度区呈现白色核心，低浓度区呈现饱满紫色
    vec3 base_rgb = mix(trail_color.rgb, glow_color.rgb, tex_color.r * 0.7);
    vec3 final_color = base_rgb * emission_intensity;
    
    // 5. 渲染输出
    ALBEDO = final_color;
    ALPHA = clamp(final_alpha, 0.0, 1.0);
}

动态拖尾网格生成脚本 (DynamicTrail3D)

挂载于 MeshInstance3D 节点，实时检测移动轨迹并计算生成高精度的 Ribbon Mesh

#=========================================================
# DynamicTrail3D.gd
# 挂载到一个空的 MeshInstance3D 节点上，实现酷炫的刀光或飞弹拖尾。
#=========================================================
extends MeshInstance3D
class_name DynamicTrail3D

@export var target_node: Node3D        # 要跟踪的物体 (例如：剑尖)
@export var offset_vector: Vector3 = Vector3(0, -1.2, 0) # 刀刃宽度的偏移向量
@export var trail_lifetime: float = 0.4 # 拖尾段持续时间 (秒)
@export var max_points: int = 60        # 最大样点数量
@export var min_distance: float = 0.05  # 生成新段所需的最小位移

struct TrailPoint:
	var top: Vector3
	var bottom: Vector3
	var time: float
	var alpha: float

var points: Array[TrailPoint] = []
var immediate_mesh: ImmediateMesh

func _ready() -> void:
	# 实例化 Godot 4 ImmediateMesh 用于动态渲染
	immediate_mesh = ImmediateMesh.new()
	self.mesh = immediate_mesh
	
	if not target_node:
		# 若未指定，则默认跟踪父级
		target_node = get_parent() as Node3D

func _process(delta: float) -> void:
	if not target_node:
		return
		
	# 1. 移去超时样点
	var current_time = Time.get_ticks_msec() / 1000.0
	var i = 0
	while i < points.size():
		var age = current_time - points[i].time
		if age > trail_lifetime:
			points.remove_at(i)
		else:
			# 计算插值Alpha，让旧段平滑消散
			points[i].alpha = 1.0 - (age / trail_lifetime)
			i += 1
			
	# 2. 采样当前骨骼/物体的物理坐标并判定是否加入新样点
	var current_top = target_node.global_position
	var current_bottom = target_node.global_global_transform().basis * offset_vector + current_top
	
	var should_add = true
	if points.size() > 0:
		var last_point = points[points.size() - 1]
		var dist = last_point.top.distance_to(current_top)
		if dist < min_distance:
			should_add = false
			
	if should_add:
		var new_point = TrailPoint.new()
		new_point.top = to_local(current_top)
		new_point.bottom = to_local(current_bottom)
		new_point.time = current_time
		new_point.alpha = 1.0
		points.append(new_point)
		
		# 限制最大网格面数
		if points.size() > max_points:
			points.remove_at(0)
			
	# 3. 动态重构三角面片 (Ribbon Mesh)
	_rebuild_trail_mesh()

func _rebuild_trail_mesh() -> void:
	immediate_mesh.clear_surfaces()
	
	if points.size() < 2:
		return
		
	immediate_mesh.surface_begin(Mesh.PRIMITIVE_TRIANGLE_STRIP)
	
	for idx in range(points.size()):
		var pt = points[idx]
		# 将索引转化为沿着整个拖尾的横向 UV 坐标 (0.0 表示尾端, 1.0 表示尖端)
		var u = float(idx) / float(points.size() - 1)
		
		# 写入上边缘顶点 (UV: [U, 1.0])
		immediate_mesh.surface_set_color(Color(1.0, 1.0, 1.0, pt.alpha))
		immediate_mesh.surface_set_uv(Vector2(u, 1.0))
		immediate_mesh.surface_add_vertex(pt.top)
		
		# 写入下边缘顶点 (UV: [U, 0.0])
		immediate_mesh.surface_set_color(Color(1.0, 1.0, 1.0, pt.alpha))
		immediate_mesh.surface_set_uv(Vector2(u, 0.0))
		immediate_mesh.surface_add_vertex(pt.bottom)
		
	immediate_mesh.surface_end()
