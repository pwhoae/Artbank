// 1. 窗戶樞軸 (Pivot)Tween 旋轉 (GDScript)
var tween = create_tween().set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_IN_OUT)
var target_y = initial_rotation.y + (90.0 if is_open else 0.0)
tween.tween_property(window_pivot, "rotation_degrees:y", target_y, 0.8)
// 2. 窗簾頂點風動 Shader (GDShader)
shader_type spatial;
render_mode cull_disabled;

uniform float wind_speed = 2.5;
uniform float wind_strength = 0.18;

void vertex() {
    float factor = pow(UV.y, 1.5); // 越靠近底部擺動越大，頂部固定
    float wave = sin(TIME * wind_speed + VERTEX.y * 3.0 + VERTEX.x) * wind_strength;
    VERTEX.z += wave * factor;
}
