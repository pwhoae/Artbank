GDScript 邏輯重構與語法優化指南

在 Godot 中，透過 `傳送門.get_children()` 獲取所有子節點位置並存入 `player_傳送門_array` 是非常經典的做法。為了實現 **「任意傳送面板」**，我們需要：

  • 動態生成 UI 按鈕：遍歷傳送門陣列，為每個門生成一個 `Button` 節點。
  • 閉包與訊號綁定 (Signal Binding)：將門的索引與座標傳遞給按鈕的 `pressed` 訊號。
  • 平滑傳送與畫面漸變：傳送時加上 Fade Out / Fade In 視覺過場，避免玩家視覺突兀。

Godot 4.x - 完整 TeleportPanel.gd

extends Control

# 節點引用 (請依專案節點樹調整路徑)
@onready var portal_container = $"../傳送門"
@onready var player = $"../Player"
@onready var grid_container = $"Panel/ScrollContainer/GridContainer"

# 儲存傳送門座標陣列與資料
var player_傳送門_array: Array[Vector2] = []

func _ready() -> void:
	update_portal_data()
	generate_teleport_panel()

# 1. 取得所有傳送門座標
func update_portal_data() -> void:
	player_傳送門_array.clear()
	for portal in portal_container.get_children():
		player_傳送門_array.append(portal.global_position)
	print("已載入傳送點陣列: ", player_傳送門_array)

# 2. 動態生成 UI 傳送面板按鈕
func generate_teleport_panel() -> void:
	# 清除舊UI內容
	for child in grid_container.get_children():
		child.queue_free()
		
	# 遍歷門點動態生成 Button
	var children = portal_container.get_children()
	for i in range(children.size()):
		var portal_node = children[i]
		var pos = player_傳送門_array[i]
		var portal_name = portal_node.name
		
		var btn = Button.new()
		btn.text = "%d. %s\n(%d, %d)" % [i + 1, portal_name, pos.x, pos.y]
		btn.custom_minimum_size = Vector2(160, 60)
		
		# Godot 4 閉包綁定按鈕點擊事件
		var target_index = i
		btn.pressed.connect(func(): teleport_to(target_index))
		
		grid_container.add_child(btn)

# 3. 執行角色傳送
func teleport_to(portal_index: int) -> void:
	if portal_index >= 0 and portal_index < player_傳送門_array.size():
		var target_pos = player_傳送門_array[portal_index]
		
		# 設定玩家座標
		player.global_position = target_pos
		print("玩家已傳送至傳送門 [", portal_index, "]: ", target_pos)
		
		# 關閉傳送面板
		hide()

進階提醒與最佳實踐

1. 名稱自訂：可在門點節點上新增導出變數 `export var portal_display_name: String`，顯示自訂選單中文名稱。
2. 畫面過場：建議搭配 `Tween` 製作 `CanvasLayer` 的 `modulate:a` 透明度淡入淡出。
