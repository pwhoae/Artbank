shader_type canvas_item;

// Godot 4 - 2D 角色回血 Shader (2D Character Heal Shader)
// 包含上升波纹、星光粒子 (Sparkles) 以及动态边缘闪烁效果

uniform vec4 heal_color : source_color = vec4(0.2, 1.0, 0.4, 1.0);
uniform float heal_progress : hint_range(0.0, 1.0) = 0.0; // 控制回血动画进度 (0:关闭, 1:最大)
uniform float pulse_speed : hint_range(0.5, 10.0) = 4.0;  // 波动速度
uniform float wave_density : hint_range(5.0, 40.0) = 15.0; // 上升波纹密度
uniform float glow_intensity : hint_range(0.0, 5.0) = 2.5; // 发光亮度

// 伪随机数生成器 (用于粒子的闪烁)
float random(vec2 uv) {
    return fract(sin(dot(uv.xy, vec2(12.9898, 78.233))) * 43758.5453123);
}

void fragment() {
    vec4 tex_color = texture(TEXTURE, UV);
    
    // 仅在有像素Alpha透明度且在回血状态时计算
    if (tex_color.a > 0.01 && heal_progress > 0.0) {
        // 1. 上升的纵向波纹 (Vertical Wave)
        float wave = sin((UV.y * wave_density) + (TIME * pulse_speed)) * 0.5 + 0.5;
        
        // 2. 从下往上推移的高度掩码 (Height Mask)
        float height_mask = smoothstep(1.0 - heal_progress - 0.2, 1.0 - heal_progress, UV.y);
        
        // 3. 上升的星光粒子 (Sparkles Noise)
        vec2 sparkle_uv = UV * 12.0 - vec2(0.0, TIME * (pulse_speed * 0.5));
        float sparkle = step(0.88, random(floor(sparkle_uv)));
        
        // 4. 组合特效强度
        float final_glow = (wave * 0.5 + sparkle * 0.5) * height_mask * glow_intensity;
        
        // 5. 整体呼吸脉冲闪光
        float pulse = (sin(TIME * pulse_speed * 1.5) * 0.5 + 0.5) * heal_progress;
        
        // 混合原图颜色与绿光
        vec3 blended_rgb = mix(tex_color.rgb, heal_color.rgb * glow_intensity, final_glow * heal_progress);
        blended_rgb += heal_color.rgb * pulse * 0.25;
        
        COLOR = vec4(blended_rgb, tex_color.a);
    } else {
        COLOR = tex_color;
    }
}

shader_type spatial;
render_mode blend_mix, depth_draw_opaque, cull_back, diffuse_burley, specular_schlick_ggx;

// Godot 4 - 3D 角色回血 Shader (3D Character Heal Shader)
// 包含 菲涅尔(Fresnel) 边缘发光、上升波纹以及 EMISSION 放射光效果

uniform sampler2D albedo_texture : source_color, filter_linear_mipmap;
uniform vec4 base_color : source_color = vec4(1.0, 1.0, 1.0, 1.0);

// 回血参数
uniform vec4 heal_color : source_color = vec4(0.2, 1.0, 0.4, 1.0);
uniform float heal_progress : hint_range(0.0, 1.0) = 0.0;
uniform float pulse_speed : hint_range(0.5, 10.0) = 4.0;
uniform float wave_density : hint_range(5.0, 40.0) = 15.0;
uniform float glow_intensity : hint_range(0.0, 10.0) = 3.0;
uniform float fresnel_power : hint_range(0.5, 8.0) = 3.0;

void fragment() {
    // 基础材质采样
    vec4 tex = texture(albedo_texture, UV) * base_color;
    ALBEDO = tex.rgb;
    
    if (heal_progress > 0.001) {
        // 1. 3D 菲涅尔边缘光 (Fresnel Rim)
        float fresnel = pow(1.0 - clamp(dot(NORMAL, VIEW), 0.0, 1.0), fresnel_power);
        
        // 2. 沿模型 height/UV 轴上升的能量波纹
        float wave = sin(UV.y * wave_density - TIME * pulse_speed) * 0.5 + 0.5;
        
        // 3. 脉冲频率
        float pulse = sin(TIME * pulse_speed * 1.5) * 0.3 + 0.7;
        
        // 4. 最终回血强度的综合计算
        float heal_factor = (fresnel * 1.5 + wave * 0.5) * pulse * heal_progress;
        
        // 赋予 EMISSION 自发光，在 Godot WorldEnvironment 中搭配 WorldEnvironment Glow 效果极佳
        EMISSION = heal_color.rgb * heal_factor * glow_intensity;
        
        // 稍微将基础颜色向回血颜色微调
        ALBEDO = mix(ALBEDO, heal_color.rgb, heal_factor * 0.25);
    }
}

GDScript 使用说明：通过 `Tween` 平滑过渡动画控制 Shader 参数。

# HealEffectController.gd
# 挂载在角色或控制脚本上，用于在玩家吃到血包或受到治疗时触发回血视觉特效
extends Node

@export_category("Heal Shader Options")
## 角色节点 (2D: Sprite2D / 3D: MeshInstance3D)
@export var character_node: Node2D # 如果是 3D 换为 GeometryInstance3D
@export var heal_duration: float = 1.2 # 回血特效持续时间(秒)
@export var heal_color: Color = Color(0.2, 1.0, 0.4, 1.0)

var _material: ShaderMaterial
var _tween: Tween

func _ready() -> void:
    if character_node:
        # 获取节点的 ShaderMaterial
        _material = character_node.material as ShaderMaterial
        if _material:
            # 初始化将 progress 设为 0
            _material.set_shader_parameter("heal_progress", 0.0)
            _material.set_shader_parameter("heal_color", heal_color)

## 触发回血效果函数
func play_heal_effect() -> void:
    if not _material:
        push_warning("未找到 ShaderMaterial，请先在 Inspecter 赋予材质！")
        return
        
    # 如果上一次动画未结束，停止原 Tween
    if _tween and _tween.is_running():
        _tween.kill()
        
    _tween = create_tween().set_parallel(false)
    
    # 阶段 1: 瞬间迅速提升 heal_progress 至 1.0
    _tween.tween_property(_material, "shader_parameter/heal_progress", 1.0, 0.25)\
        .set_trans(Tween.TRANS_CUBIC)\
        .set_ease(Tween.EASE_OUT)
        
    # 阶段 2: 渐隐 fading out 回 0.0
    _tween.tween_property(_material, "shader_parameter/heal_progress", 0.0, heal_duration - 0.25)\
        .set_trans(Tween.TRANS_QUAD)\
        .set_ease(Tween.EASE_IN_OUT)

## 演示测试：按 H 键触发回血
func _unhandled_input(event: InputEvent) -> void:
    if event is InputEventKey and event.pressed and event.keycode == KEY_H:
        play_heal_effect()
