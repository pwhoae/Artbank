## 💻 Godot 4 核心控制腳本 (`space_capsule_ui.gd`)

```gdscript
extends Control

# 綁定音效播放器節點
@onready var boot_up_player = $AudioPlayers/BootUpPlayer
@onready var click_player = $AudioPlayers/ClickPlayer
@onready var slider_gen_player = $AudioPlayers/SliderGeneratorPlayer
@onready var alert_player = $AudioPlayers/AlertPlayer

# 綁定 UI 組件節點
@onready var unlock_button = $UI_Elements/UnlockButton
@onready var parameter_slider = $UI_Elements/ParameterSlider
@onready var alert_button = $UI_Elements/AlertButton
@onready var matrix_buttons = $UI_Elements/MatrixButtons

# 音頻生成器相關變數 (用於滑桿連續音)
var playback: AudioStreamGeneratorPlayback
var sample_rate: float = 44100.0
var phase: float = 0.0
var target_frequency: float = 0.0
var is_sliding: bool = false

func _ready():
	# 1. 初始化音頻生成器 (Slider 專用)
	_setup_audio_generator()
	
	# 2. 綁定 UI 訊號 (Signals) 到對應的邏輯與音效
	unlock_button.pressed.connect(_on_unlock_button_pressed)
	parameter_slider.value_changed.connect(_on_slider_value_changed)
	parameter_slider.drag_started.connect(func(): is_sliding = true)
	parameter_slider.drag_ended.connect(func(_changed): is_sliding = false)
	alert_button.pressed.connect(_on_alert_button_pressed)
	
	# 自動遍歷並綁定矩陣切換區的所有按鈕
	for button in matrix_buttons.get_children():
		if button is Button:
			button.pressed.connect(_on_matrix_button_pressed)

func _process(delta):
	# 當用戶在滑動滑桿時，即時動態生成音頻正弦波
	if is_sliding and playback:
		_fill_audio_buffer()

# --- A. 系統解鎖區 (Boot Up) ---
func _on_unlock_button_pressed():
	print("系統解鎖中...")
	# 視覺微動畫：輕微放大並漸變
	var tween = create_tween().set_parallel(true)
	tween.tween_property(unlock_button, "scale", Vector2(1.05, 1.05), 0.1)
	tween.tween_property(unlock_button, "modulate", Color(0.7, 0.9, 1.0), 0.1)
	tween.chain().tween_property(unlock_button, "scale", Vector2(1.0, 1.0), 0.1)
	
	# 播放啟動音效 (低頻環境氣流 -> 升調合成音)
	if not boot_up_player.playing:
		boot_up_player.play()

# --- B. 矩陣切換區 (Click/Tap) ---
func _on_matrix_button_pressed():
	print("矩陣按鈕點擊")
	# 極短清脆的微動開關反饋音 (延遲控制在 10ms 以內)
	click_player.play()

# --- C. 參數調節區 (Scroll/Slide 動態生成) ---
func _setup_audio_generator():
	# 動態創建一個音頻生成器串流，長度設為 0.5 秒即可（會循環填充）
	var generator = AudioStreamGenerator.new()
	generator.mix_rate = sample_rate
	generator.buffer_length = 0.5
	
	slider_gen_player.stream = generator
	slider_gen_player.play()
	# 獲取播放緩衝區控制權
	playback = slider_gen_player.get_stream_playback()

func _on_slider_value_changed(value: float):
	# 根據滑桿數值 (假設滑桿範圍是 0 ~ 100)，動態映射到音頻頻率 (200Hz ~ 1200Hz)
	# 實現類似復古收音機或能量蓄積的顆粒感升調
	target_frequency = remap(value, parameter_slider.min_value, parameter_slider.max_value, 200.0, 1200.0)

func _fill_audio_buffer():
	# 填充音頻緩衝區，產生絲滑的頻率變化
	var frames_available = playback.get_frames_available()
	for i in range(frames_available):
		# 計算正弦波
		var sample = sin(phase * TAU)
		# 加上一點微弱的“顆粒感/方波調製”讓它更具科幻感
		sample += (randf() - 0.5) * 0.05 
		
		playback.push_frame(Vector2(sample, sample) * 0.2) # 0.2 為音量控制
		
		# 根據目標頻率更新相位步進
		var increment = target_frequency / sample_rate
		phase = wrapf(phase + increment, 0.0, 1.0)

# --- D. 警報與通知 (Alert) ---
func _on_alert_button_pressed():
	print("緊急警報觸發！")
	# 視覺特效：螢幕紅光閃爍
	var tween = create_tween()
	tween.tween_property(self, "modulate", Color(1.5, 0.5, 0.5), 0.1) # 螢幕泛紅 (利用HDR過曝)
	tween.tween_property(self, "modulate", Color(1, 1, 1), 0.3)
	
	# 播放高穿透力的雙音調交替警告音
	if not alert_player.playing:
		alert_player.play()

```

## 🎛️ Godot 音效優化小撇步 (Audio Spec Setup)

為了完美呈現你策劃案中要求的**高質感、極致流暢度**，請在 Godot 編輯器中做以下調整：

1. **零延遲設定 (Audio Latency):**
* 前往 `項目 (Project) -> 項目設置 (Project Settings) -> Audio (音頻)`。
* 將 **Output Latency (輸出延遲)** 調低（例如從默認的 15 降到 **10** 或 **8**），這能確保點擊按鈕時，視覺與聽覺在 10ms 內完美同步。


2. **警報音的迴音效果 (Alert Echo):**
* 在 Godot 底部點擊 **Audio** 面板，為 `Master` 總線或新創一個 `Alert` 總線添加一個 **`AudioEffectReverb` (混響)** 或 **`AudioEffectDelay` (延遲)** 效果器。
* 將 `AlertPlayer` 的 `Bus` 設為該總線，就能直接做出策劃案中要求的“穿透力雙音調且帶有微弱回音”的現代高科技感。


3. **滑桿的音頻動態:**
* 腳本中使用的 `AudioStreamGeneratorPlayback` 可以做到完美的跟隨。當你手指（或滑鼠）拖動 `HSlider` 時，音頻頻率會跟著手速毫無延遲地變高或變低。

# 🚀 主脚本（直接可用）

挂在 `Main` 上

```gdscript
extends Control

@onready var unlock_btn = $UnlockButton
@onready var slider = $Slider
@onready var alert_btn = $AlertButton

@onready var audio_click = $Audio_Click
@onready var audio_boot = $Audio_Boot
@onready var audio_alert = $Audio_Alert
@onready var audio_slider = $Audio_Slider

var is_unlocked = false

func _ready():
    # 解锁按钮
    unlock_btn.pressed.connect(_on_unlock_pressed)

    # 普通按钮
    for btn in $ButtonPanel.get_children():
        btn.pressed.connect(_on_click_button)

    # 滑条
    slider.value_changed.connect(_on_slider_changed)

    # 警报
    alert_btn.pressed.connect(_on_alert_pressed)


# =========================
# A. 解锁系统（Boot Sound）
# =========================
func _on_unlock_pressed():
    if is_unlocked:
        return

    is_unlocked = true

    audio_boot.play()

    # UI 动画（增强质感）
    unlock_btn.scale = Vector2(1.2, 1.2)
    await get_tree().create_timer(0.1).timeout
    unlock_btn.scale = Vector2(1, 1)


# =========================
# B. 机械按钮（Click Sound）
# =========================
func _on_click_button():
    audio_click.play()

    var btn = get_viewport().gui_get_focus_owner()
    if btn:
        btn.scale = Vector2(0.9, 0.9)
        await get_tree().create_timer(0.05).timeout
        btn.scale = Vector2(1, 1)


# =========================
# C. 滑条（动态音频）
# =========================
func _on_slider_changed(value):
    if not audio_slider.playing:
        audio_slider.play()

    # 核心：用 pitch 模拟频率变化
    var pitch = lerp(0.5, 2.0, value / slider.max_value)
    audio_slider.pitch_scale = pitch


# =========================
# D. 警报系统（Alert Sound）
# =========================
func _on_alert_pressed():
    audio_alert.play()

    # 闪烁效果（警告 UI）
    flash_alert()


func flash_alert():
    for i in range(4):
        alert_btn.modulate = Color(1, 0.3, 0.3)
        await get_tree().create_timer(0.1).timeout
        alert_btn.modulate = Color(1, 1, 1)
        await get_tree().create_timer(0.1).timeout
```
