高級感 UI 核心設計手法

影片中展現的 Godot 擬物化 UI（Skeuomorphic / Neumorphism）主要透過以下三個層次實現：

  • 1. 精細的雙向立體投影：左上高光與右下深色投影的相互堆疊（凸起按鈕與內陷凹槽），為 2D 介面創造驚人的物理立體縱深。
  • 2. 著色器（Shaders）繪製刻度：不使用靜態圖片，而是用 GPU 逐像素渲染高精度的弧形刻度、虛線與發光點，保證動態縮放不模糊。
  • 3. 數值平滑插值（Tweening）：在拖拽或設定數值時，數值和指標不會瞬間跳變，而是透過 GDScript 的 lerp() 進行緩動過渡。

應用於 TextureRect 節點，可完美渲染出帶發光虛線的圓環進度條。 
shader_type canvas_item;

uniform float value : hint_range(0.0, 1.0) = 0.45; // 目前進度 (0.0 到 1.0)
uniform vec4 track_color : source_color = vec4(0.1, 0.12, 0.15, 1.0); // 底色
uniform vec4 progress_color : source_color = vec4(0.0, 0.7, 1.0, 1.0); // 進度發光顏色
uniform float thickness : hint_range(0.01, 0.2) = 0.06; // 圓環厚度
uniform float num_dashes : hint_range(10.0, 100.0) = 48.0; // 刻度線數量
uniform float gap_ratio : hint_range(0.1, 0.9) = 0.4; // 刻度虛線間距比例

void fragment() {
    vec2 uv = UV - vec2(0.5);
    float dist = length(uv);
    
    // 限制在圓環厚度範圍內
    float inner_radius = 0.4 - thickness;
    float outer_radius = 0.4;
    float ring_mask = smoothstep(inner_radius - 0.005, inner_radius, dist) * (1.0 - smoothstep(outer_radius, outer_radius + 0.005, dist));
    
    // 計算角度分量
    float angle = atan(uv.y, uv.x) + PI; // 對齊到 0 到 2PI
    float normalized_angle = angle / (2.0 * PI);
    
    // 建立起始缺口 (模仿影片下方的部分空缺，例如 240度環繞)
    // 假設留空底部的 60度 (約合 0.83 的扇形)
    float active_range = 0.83; 
    float offset_angle = mod(normalized_angle + 0.58, 1.0); // 旋轉起點至頂部偏下
    
    if (offset_angle > active_range) {
        discard; // 裁切掉多餘的環狀缺口
    }
    
    float range_angle = offset_angle / active_range;
    
    // 虛線刻度遮罩
    float dash_val = fract(range_angle * num_dashes);
    float dash_mask = step(gap_ratio, dash_val);
    
    // 計算最終色彩
    vec4 final_color = track_color;
    if (range_angle <= value) {
        // 渲染發光進度色，並在邊緣添加發光感
        float glow = 1.0 + 0.5 * sin(TIME * 2.0);
        final_color = progress_color * glow;
    }
    
    COLOR = final_color * ring_mask * dash_mask;
}
掛載於控制節點，處理滑動/旋轉，並提供極致流暢的彈簧緩動。 
extends TextureRect

# 信號通知
signal value_changed(new_value)

@export var min_value: float = 0.0
@export var max_value: float = 100.0
@export var current_value: float = 45.0:
	set(val):
		current_value = clamp(val, min_value, max_value)
		_update_ui()

# 用於平滑視覺過渡
var visual_value: float = 45.0
var is_dragging: bool = false
var drag_start_angle: float = 0.0
var value_on_drag_start: float = 0.0

@onready var label: Label = $ValueLabel

func _ready() -> void:
	visual_value = current_value
	_update_ui()

func _process(delta: float) -> void:
	# 物理平滑過渡 (Lerp)，達成高級感的核心
	if not is_equal_approx(visual_value, current_value):
		visual_value = lerp(visual_value, current_value, 15.0 * delta)
		# 將進度值平滑傳遞給 Shader
		(material as ShaderMaterial).set_shader_parameter("value", visual_value / max_value)
		if label:
			label.text = str(round(visual_value))

func _gui_input(event: InputEvent) -> void:
	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_LEFT:
			if event.pressed:
				is_dragging = true
				drag_start_angle = _get_mouse_angle(event.position)
				value_on_drag_start = current_value
			else:
				is_dragging = false
				
	elif event is InputEventMouseMotion and is_dragging:
		var current_angle = _get_mouse_angle(event.position)
		var angle_diff = current_angle - drag_start_angle
		
		# 邊界防止突變處理
		if angle_diff > PI: angle_diff -= 2.0 * PI
		elif angle_diff < -PI: angle_diff += 2.0 * PI
		
		# 1 弧度代表 25 單位數值變化
		var sensitivity: float = 25.0 
		self.current_value = value_on_drag_start + (angle_diff * sensitivity)
		value_changed.emit(current_value)

func _get_mouse_angle(mouse_pos: Vec2) -> float:
	var center = size / 2.0
	var dir = mouse_pos - center
	return atan2(dir.y, dir.x)

func _update_ui() -> void:
	# 靜態更新備份
	if not is_inside_tree(): return
	if label and not is_dragging:
		label.text = str(round(current_value))
利用片元著色器精確繪製擬物面板的軟陰影（Soft Shadow）與內凹槽質感。 
shader_type canvas_item;

uniform vec4 base_color : source_color = vec4(0.1, 0.11, 0.13, 1.0);
uniform float corner_radius : hint_range(0.01, 0.5) = 0.1;
uniform bool is_inset = false; // 是否為內凹槽

// 用於計算圓角矩形距離場
float sd_round_rect(vec2 p, vec2 b, float r) {
    vec2 q = abs(p) - b + r;
    return min(max(q.x, q.y), 0.0) + length(max(q, 0.0)) - r;
}

void fragment() {
    vec2 size = 1.0 / TEXTURE_PIXEL_SIZE;
    vec2 uv = (UV - vec2(0.5)) * size;
    vec2 half_size = size * 0.5;
    
    float radius = min(half_size.x, half_size.y) * corner_radius;
    float dist = sd_round_rect(uv, half_size, radius);
    
    // 裁剪外圍圓角以外的部分
    if (dist > 0.0) {
        discard;
    }
    
    vec4 final_color = base_color;
    
    // 精準微光陰影疊加
    if (is_inset) {
        // 內陷效果：左上深色，右下亮色微光
        float shadow_mask = smoothstep(-25.0, 0.0, dist);
        vec2 shadow_dir = normalize(vec2(1.0, 1.0));
        float light_factor = dot(normalize(uv), shadow_dir);
        
        if (light_factor > 0.0) {
            // 右下邊緣亮光
            final_color += vec4(0.08, 0.09, 0.12, 0.0) * light_factor * shadow_mask;
        } else {
            // 左上邊緣深色陰影
            final_color -= vec4(0.08, 0.08, 0.09, 0.0) * abs(light_factor) * shadow_mask;
        }
    } else {
        // 凸起效果：邊緣細緻高光
        float edge_width = 1.5;
        if (dist > -edge_width) {
            float edge_factor = smoothstep(-edge_width, 0.0, dist);
            // 頂部與左側有微弱高光，營造光源自左上角照下的立體感
            if (uv.y < 0.0 || uv.x < 0.0) {
                final_color += vec4(0.12, 0.14, 0.18, 0.0) * edge_factor;
            }
        }
    }
    
    COLOR = final_color;
}
