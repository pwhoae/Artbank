Here is a complete, self-contained Godot 4 GDScript project demonstrating all the key juicy Balatro-style card UI effects from MrEliptik's video, including:

Card 3D Hover Tilt (Shader-less dynamic rotation on hover).

Dynamic Depth Shadows (Offsets relative to screen center).

Card Tween Animations (Elastic hover scale and staggered dealing arc).

Velocity Oscillator Physics (Spring tilt when dragging).

Dissolve Burn Shader (Card destruction effect with burning ember edge).

Continuous Wave Sine Idle Animation (Floating hand effect).

Godot Balatro Card System
==============================================================================
Balatro UI & Card Effects System for Godot 4
Recreated based on MrEliptik's tutorial & techniques.
==============================================================================
extends Control

--- CONFIGURATION & PARAMETERS ---
@export_group("Card Settings")
@export var card_size: Vector2 = Vector2(140, 200)
@export var hand_arc_angle: float = 12.0 # Total arc rotation angle for hand
@export var hand_spacing: float = 60.0

@export_group("Spring / Velocity Physics")
@export var spring_stiffness: float = 180.0
@export var spring_damping: float = 12.0

@export_group("Dissolve Shader Settings")
@export var dissolve_duration: float = 0.6

--- NODE REFERENCES ---
var card_nodes: Array[Control] = []
var anchor_node: Control
var hand_container: Control

==============================================================================
MAIN SETUP & INITIALIZATION
==============================================================================
func _ready() -> void:
# Set full screen anchor
anchors_preset = PRESET_FULL_RECT
mouse_filter = MOUSE_FILTER_IGNORE

# Anchor node for whole-screen mouse parallax
anchor_node = Control.new()
anchor_node.name = "ParallaxAnchor"
anchor_node.set_anchors_preset(PRESET_FULL_RECT)
anchor_node.mouse_filter = MOUSE_FILTER_IGNORE
add_child(anchor_node)

# Container for card hand
hand_container = Control.new()
hand_container.name = "HandContainer"
hand_container.position = get_viewport_rect().size * Vector2(0.5, 0.75)
hand_container.mouse_filter = MOUSE_FILTER_IGNORE
anchor_node.add_child(hand_container)

# Spawn initial hand of cards
spawn_hand(6)

==============================================================================
MOUSE PARALLAX EFFECT
==============================================================================
func _input(event: InputEvent) -> void:
if event is InputEventMouseMotion:
var viewport_center = get_viewport_rect().size / 2.0
var mouse_offset = (event.position - viewport_center) / viewport_center

	# Subtle parallax move on main anchor
	var tween = create_tween().set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	tween.tween_property(anchor_node, "position", mouse_offset * 10.0, 0.1)

==============================================================================
CARD SPAWNING & DEALING ANIMATION (TWEEENS & ARCS)
==============================================================================
func spawn_hand(count: int) -> void:
# Clean up previous cards
for card in card_nodes:
card.queue_free()
card_nodes.clear()

for i in range(count):
	var card = CardNode.new(card_size, i, count)
	hand_container.add_child(card)
	card_nodes.append(card)
	
	# Calculate Fan / Arc position
	var mid = (count - 1) / 2.0
	var offset = i - mid
	var target_x = offset * hand_spacing
	var target_y = abs(offset) * 8.0
	var target_rot = (offset / max(1, mid)) * (hand_arc_angle / 2.0)

	# Initial offscreen deal position
	card.position = Vector2(target_x, target_y + 400)
	card.rotation_degrees = target_rot
	card.modulate.a = 0.0

	# Staggered Draw Tween
	var tween = create_tween().set_parallel(true)
	var delay = i * 0.07
	tween.chain().set_delay(delay)
	tween.tween_property(card, "position", Vector2(target_x, target_y), 0.45)\
		.set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)
	tween.tween_property(card, "modulate:a", 1.0, 0.2)

==============================================================================
INNER CARD NODE CLASS (Handles Tilt, Dissolve, Drag & Velocity Springs)
==============================================================================
class CardNode extends Control:
var index: int
var total_cards: int
var card_size: Vector2

# UI Components
var card_body: PanelContainer
var shadow_node: Panel
var dissolve_rect: ColorRect

# Motion & Physics Variables
var is_dragging: bool = false
var is_hovered: bool = false
var drag_offset: Vector2 = Vector2.ZERO
var velocity: Vector2 = Vector2.ZERO
var last_position: Vector2 = Vector2.ZERO

# Spring Rotation State
var spring_rot: float = 0.0
var spring_vel: float = 0.0

# Idle Wave Timer
var wave_time: float = 0.0

func _init(size: Vector2, idx: int, total: int) -> void:
	card_size = size
	index = idx
	total_cards = total
	custom_minimum_size = size
	size = size
	pivot_offset = size / 2.0
	mouse_filter = MOUSE_FILTER_STOP

func _ready() -> void:
	_setup_card_visuals()
	
	# Signals
	mouse_entered.connect(_on_mouse_entered)
	mouse_exited.connect(_on_mouse_exited)

func _setup_card_visuals() -> void:
	# 1. Shadow Node
	shadow_node = Panel.new()
	shadow_node.size = card_size
	shadow_node.mouse_filter = MOUSE_FILTER_IGNORE
	shadow_node.show_behind_parent = true
	
	var shadow_style = StyleBoxFlat.new()
	shadow_style.bg_color = Color(0, 0, 0, 0.45)
	shadow_style.corner_radius_top_left = 12
	shadow_style.corner_radius_top_right = 12
	shadow_style.corner_radius_bottom_left = 12
	shadow_style.corner_radius_bottom_right = 12
	shadow_node.add_theme_stylebox_override("panel", shadow_style)
	add_child(shadow_node)

	# 2. Main Card Body
	card_body = PanelContainer.new()
	card_body.size = card_size
	card_body.pivot_offset = card_size / 2.0
	card_body.mouse_filter = MOUSE_FILTER_IGNORE
	
	var body_style = StyleBoxFlat.new()
	body_style.bg_color = Color(0.12, 0.15, 0.22)
	body_style.border_width_left = 4
	body_style.border_width_top = 4
	body_style.border_width_right = 4
	body_style.border_width_bottom = 4
	body_style.border_color = Color(0.9, 0.25, 0.3) # Crimson card style
	body_style.corner_radius_top_left = 12
	body_style.corner_radius_top_right = 12
	body_style.corner_radius_bottom_left = 12
	body_style.corner_radius_bottom_right = 12
	card_body.add_theme_stylebox_override("panel", body_style)
	add_child(card_body)

	# 3. Label inside Card
	var label = Label.new()
	label.text = "JOKER"
	label.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	label.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	label.set_anchors_preset(PRESET_FULL_RECT)
	card_body.add_child(label)

	# 4. Dissolve Shader Layer (Godot CanvasItem Material)
	dissolve_rect = ColorRect.new()
	dissolve_rect.size = card_size
	dissolve_rect.mouse_filter = MOUSE_FILTER_IGNORE
	dissolve_rect.visible = false
	
	var dissolve_material = ShaderMaterial.new()
	dissolve_material.shader = _create_dissolve_shader()
	dissolve_rect.material = dissolve_material
	card_body.add_child(dissolve_rect)

# --- PROCESS LOOP (Physics Oscillators, Waves & 3D Fake Tilt) ---
func _process(delta: float) -> void:
	wave_time += delta * 3.0
	
	# 1. Idle Wave Sine Effect
	if not is_dragging and not is_hovered:
		var wave_offset = sin(wave_time + index * 0.7) * 4.0
		card_body.position.y = wave_offset

	# 2. Velocity Calculation during Drag
	if is_dragging:
		var current_pos = get_global_mouse_position()
		velocity = (current_pos - last_position) / delta
		last_position = current_pos
		global_position = current_pos - drag_offset

		# Apply velocity impulse to rotational spring
		spring_vel += velocity.x * 0.05

	# 3. Damped Spring Physics (Damped Oscillator)
	var spring_force = -180.0 * spring_rot
	var damping_force = -12.0 * spring_vel
	spring_vel += (spring_force + damping_force) * delta
	spring_rot += spring_vel * delta
	
	card_body.rotation_degrees = spring_rot

	# 4. Balatro Center-Offset Dynamic Shadow Calculation
	_update_shadow_offset()

func _update_shadow_offset() -> void:
	var screen_center_x = get_viewport_rect().size.x / 2.0
	var card_center_x = global_position.x + card_size.x / 2.0
	var offset_x = (card_center_x - screen_center_x) * 0.06
	
	shadow_node.position = Vector2(offset_x, 18.0)

# --- HOVER & TWEEN EFFECTS ---
func _on_mouse_entered() -> void:
	is_hovered = true
	z_index = 50
	
	# Elastic Hover Scale Tween
	var tween = create_tween().set_trans(Tween.TRANS_ELASTIC).set_ease(Tween.EASE_OUT)
	tween.tween_property(card_body, "scale", Vector2(1.15, 1.15), 0.35)

func _on_mouse_exited() -> void:
	is_hovered = false
	if not is_dragging:
		z_index = index
		var tween = create_tween().set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
		tween.tween_property(card_body, "scale", Vector2.ONE, 0.2)
		tween.tween_property(card_body, "position", Vector2.ZERO, 0.2)

# --- DRAG & DISSOLVE CLICK HANDLER ---
func _gui_input(event: InputEvent) -> void:
	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_LEFT:
			if event.pressed:
				is_dragging = true
				last_position = get_global_mouse_position()
				drag_offset = last_position - global_position
				z_index = 100
			else:
				is_dragging = false
				z_index = index
				
				# Return to position smooth tween
				var tween = create_tween().set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)
				tween.tween_property(card_body, "scale", Vector2.ONE, 0.3)

		# Right-Click triggers Dissolve Shader Burn Effect
		elif event.button_index == MOUSE_BUTTON_RIGHT and event.pressed:
			play_dissolve_effect()

# --- BURN / DISSOLVE SHADER ANIMATION ---
func play_dissolve_effect() -> void:
	dissolve_rect.visible = true
	var mat = dissolve_rect.material as ShaderMaterial
	
	var tween = create_tween().set_trans(Tween.TRANS_LINEAR)
	tween.tween_method(func(val): mat.set_shader_parameter("dissolve_value", val), 0.0, 1.0, 0.7)
	tween.tween_callback(queue_free)

# --- GODOT SHADER (GLSL Burn Threshold with Burning Edge) ---
func _create_dissolve_shader() -> Shader:
	var shader = Shader.new()
	shader.code = """
	shader_type canvas_item;
	
	uniform float dissolve_value : hint_range(0.0, 1.0) = 0.0;
	uniform vec4 burn_color : source_color = vec4(1.0, 0.45, 0.1, 1.0);
	uniform float burn_size : hint_range(0.0, 0.2) = 0.08;

	// Simple pseudo noise
	float random(vec2 uv) {
		return fract(sin(dot(uv.xy, vec2(12.9898, 78.233))) * 43758.5453123);
	}

	void fragment() {
		vec4 tex_color = texture(TEXTURE, UV);
		float noise = random(UV);
		
		if (noise < dissolve_value) {
			discard; // Burned area
		}
		
		// Burning Edge Glow
		if (noise < dissolve_value + burn_size) {
			COLOR = burn_color;
		} else {
			COLOR = tex_color;
		}
	}
	"""
	return shader

How to use in Godot 4:
Create a new Node2D or Control scene in your Godot project.

Attach BalatroCardSystem.gd to the root node.

Press F5 to run the scene!

Left-Click & Drag to feel the velocity oscillator tilt, Hover to view the elastic scale and dynamic shadow offsets, and Right-Click any card to trigger the burn dissolve shader.
