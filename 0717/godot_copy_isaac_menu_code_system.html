# ==========================================
# menu.gd - Main State Machine and Controller
# ==========================================
extends Control

# Defined states corresponding to Godot menu screens
enum MenuState {
    START_SCREEN,
    SAVE_SELECT,
    MAIN_MENU,
    CHARACTER_SELECT,
    OPTIONS
}

var current_state : MenuState = MenuState.START_SCREEN
var selected_save_slot : int = -1
var main_menu_index : int = 0
var selected_character_idx : int = 0
var current_difficulty : int = 0 # 0: Normal, 1: Hard, 2: Greed

# Node Onready paths (corresponds with Godot hierarchy)
@onready var screen_start = $ChooseSaveData/StartScreen
@onready var screen_save = $ChooseSaveData/SaveSelector
@onready var screen_main_menu = $ChooseSaveData/MainMenu
@onready var screen_char_select = $ChooseActor
@onready var menu_buttons = $ChooseSaveData/MainMenu/VBoxContainer
@onready var camera = $MenuCamera

func _ready():
    # Initialize menus hidden except the Start screen
    _reset_menu_visibility()
    _update_menu_pointers()

func _input(event):
    if event.is_action_pressed("ui_accept"):
        _handle_accept()
    elif event.is_action_pressed("ui_cancel"):
        _handle_back()
    elif event.is_action_pressed("ui_up"):
        _handle_navigation(-1, "vertical")
    elif event.is_action_pressed("ui_down"):
        _handle_navigation(1, "vertical")
    elif event.is_action_pressed("ui_left"):
        _handle_navigation(-1, "horizontal")
    elif event.is_action_pressed("ui_right"):
        _handle_navigation(1, "horizontal")

func change_state(new_state: MenuState):
    # Godot 4 Tween Transitions
    var tween = create_tween().set_parallel(true)
    tween.set_trans(Tween.TRANS_QUART).set_ease(Tween.EASE_OUT)
    
    # Animate current state away (Slide out & modulate)
    _animate_out_state(current_state, tween)
    # Animate new state in (Slide in & modulate)
    _animate_in_state(new_state, tween)
    
    current_state = new_state

func _animate_out_state(state: MenuState, tween: Tween):
    match state:
        MenuState.START_SCREEN:
            tween.tween_property(screen_start, "position:y", -500.0, 0.4)
            tween.tween_property(screen_start, "modulate:a", 0.0, 0.3)
        MenuState.SAVE_SELECT:
            tween.tween_property(screen_save, "position:y", -500.0, 0.4)
            tween.tween_property(screen_save, "modulate:a", 0.0, 0.3)
        MenuState.MAIN_MENU:
            tween.tween_property(screen_main_menu, "position:x", -800.0, 0.4)
            tween.tween_property(screen_main_menu, "modulate:a", 0.0, 0.3)
        MenuState.CHARACTER_SELECT:
            tween.tween_property(screen_char_select, "position:y", 500.0, 0.4)
            tween.tween_property(screen_char_select, "modulate:a", 0.0, 0.3)

func _animate_in_state(state: MenuState, tween: Tween):
    match state:
        MenuState.SAVE_SELECT:
            screen_save.position.y = 500.0
            screen_save.visible = true
            tween.tween_property(screen_save, "position:y", 0.0, 0.4)
            tween.tween_property(screen_save, "modulate:a", 1.0, 0.4)
        MenuState.MAIN_MENU:
            screen_main_menu.position.x = 800.0
            screen_main_menu.visible = true
            tween.tween_property(screen_main_menu, "position:x", 0.0, 0.4)
            tween.tween_property(screen_main_menu, "modulate:a", 1.0, 0.4)
        MenuState.CHARACTER_SELECT:
            screen_char_select.position.y = -500.0
            screen_char_select.visible = true
            tween.tween_property(screen_char_select, "position:y", 0.0, 0.4)
            tween.tween_property(screen_char_select, "modulate:a", 1.0, 0.4)

func _handle_accept():
    match current_state:
        MenuState.START_SCREEN:
            _play_sound("confirm")
            change_state(MenuState.SAVE_SELECT)
        MenuState.SAVE_SELECT:
            _play_sound("confirm")
            selected_save_slot = 1 # default slot mock
            change_state(MenuState.MAIN_MENU)
        MenuState.MAIN_MENU:
            _play_sound("confirm")
            if main_menu_index == 0: # "新游戏"
                change_state(MenuState.CHARACTER_SELECT)
        MenuState.CHARACTER_SELECT:
            _play_sound("game_start")
            # Load Game Level scene here!

func _handle_back():
    match current_state:
        MenuState.SAVE_SELECT:
            _play_sound("cancel")
            change_state(MenuState.START_SCREEN)
        MenuState.MAIN_MENU:
            _play_sound("cancel")
            change_state(MenuState.SAVE_SELECT)
        MenuState.CHARACTER_SELECT:
            _play_sound("cancel")
            change_state(MenuState.MAIN_MENU)

func _handle_navigation(direction: int, axis: String):
    match current_state:
        MenuState.MAIN_MENU:
            if axis == "vertical":
                main_menu_index = clampi(main_menu_index + direction, 0, 5)
                _play_sound("move")
                _update_menu_pointers()
        MenuState.CHARACTER_SELECT:
            if axis == "horizontal":
                selected_character_idx = clampi(selected_character_idx + direction, 0, 2)
                _play_sound("move")
                _update_character_ui()
            elif axis == "vertical":
                current_difficulty = clampi(current_difficulty + direction, 0, 2)
                _play_sound("move")
                _update_difficulty_ui()

func _update_menu_pointers():
    for i in range(6):
        var btn = menu_buttons.get_child(i)
        var pointer = btn.get_node("Pointer")
        pointer.visible = (i == main_menu_index)

func _reset_menu_visibility():
    screen_start.visible = true
    screen_save.visible = false
    screen_main_menu.visible = false
    screen_char_select.visible = false

# save_manager.gd - Persistence Handling File
# ==========================================
extends Node

const SAVE_PATH_BASE = "user://save_slot_"

func save_game(slot: int, data: Dictionary) -> bool:
    var file_path = SAVE_PATH_BASE + str(slot) + ".json"
    var file = FileAccess.open(file_path, FileAccess.WRITE)
    if file == null:
        printerr("Could not open save file: ", file_path)
        return false
        
    var json_string = JSON.stringify(data)
    file.store_line(json_string)
    file.close()
    return true

func load_game(slot: int) -> Dictionary:
    var file_path = SAVE_PATH_BASE + str(slot) + ".json"
    if not FileAccess.file_exists(file_path):
        return {} # Return empty if save doesn't exist
        
    var file = FileAccess.open(file_path, FileAccess.READ)
    if file == null:
        return {}
        
    var json_string = file.get_line()
    file.close()
    
    var json = JSON.new()
    var error = json.parse(json_string)
    if error == OK:
        return json.get_data()
    else:
        printerr("Failed to parse JSON file on slot ", slot)
        return {}
