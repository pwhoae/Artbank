1. CSS Design Tokens & Accessibility 

:root {
  /* Disney Motion Duration Tokens */
  --motion-duration-instant: 100ms;
  --motion-duration-fast: 200ms;
  --motion-duration-normal: 300ms;
  --motion-duration-slow: 400ms;
  --motion-duration-expressive: 500ms;

  /* Disney Adapted Easing Tokens */
  --motion-ease-standard: cubic-bezier(0.2, 0.0, 0.0, 1.0); /* Slow In/Out */
  --motion-ease-entrance: cubic-bezier(0.0, 0.0, 0.2, 1.0); /* Ease Out */
  --motion-ease-exit: cubic-bezier(0.4, 0.0, 1.0, 1.0);     /* Ease In */
  --motion-ease-bounce: cubic-bezier(0.34, 1.56, 0.64, 1.0); /* Anticipation */
}

/* Reduced Motion 无障碍统一降级 */
@media (prefers-reduced-motion: reduce) {
  *, *::before, *::after {
    animation-duration: 0.01ms !important;
    animation-iteration-count: 1 !important;
    transition-duration: 0.15s !important;
    scroll-behavior: auto !important;
  }
}

2. React / Framer Motion 

import { motion, useReducedMotion } from 'framer-motion';

// Disney Stagger List Variants
const listVariants = {
  hidden: { opacity: 0 },
  visible: {
    opacity: 1,
    transition: { staggerChildren: 0.04 } // 40ms stagger
  }
};

const itemVariants = {
  hidden: { opacity: 0, y: 16 },
  visible: {
    opacity: 1,
    y: 0,
    transition: {
      duration: 0.3, // 300ms
      ease: [0.2, 0.0, 0.0, 1.0] // Slow In/Out
    }
  }
};

export const DisneyList = ({ items }) => {
  const shouldReduceMotion = useReducedMotion();

  return (
    <motion.ul variants={listVariants} initial="hidden" animate="visible">
      {items.map((item) => (
        <motion.li key={item.id} variants={shouldReduceMotion ? {} : itemVariants}>
          {item.title}
        </motion.li>
      ))}
    </motion.ul>
  );
};
