# 挂载节点: Wheel (Node2D) | 核心算法: RayCast2D 极坐标向量角度计算与 2π/N 环形自动布局 
extends Node2D

# 节点引用
@onready var ray_cast_2d: RayCast2D = $RayCast2D
@onready var items_container: Node2D = $Items

# 配置参数
@export var radius: float = 110.0
@export var wheel_item_scene: PackedScene

var selected_item: Area2D = null
var is_open: bool = false

func _ready() -> void:
    items_container.visible = false

func _process(_delta: float) -> void:
    if not is_open:
        return
        
    # 1. 获取鼠标相对于 Wheel 节点的本地位置
    var mouse_pos = get_local_mouse_position()
    var angle = mouse_pos.angle() # 获取弧度 rad
    
    # 2. 动态改变 RayCast2D 的检测方向与距离 (500 像素)
    ray_cast_2d.target_position = Vector2(cos(angle), sin(angle)) * 500.0
    ray_cast_2d.force_raycast_update()
    
    # 3. 碰撞检测判定选中的道具
    if ray_cast_2d.is_colliding():
        var collider = ray_cast_2d.get_collider()
        if collider is Area2D:
            selected_item = collider
    else:
        selected_item = null
        
    # 4. 更新图标放大与焦点状态
    _update_selection_effect()

func _update_selection_effect() -> void:
    for item in items_container.get_children():
        if item == selected_item:
            item.update_scale(1.4) # 聚焦放大 1.4 倍
        else:
            item.update_scale(1.0) # 保持原大小

# 打开环形选择器
func open() -> void:
    if is_open: return
    is_open = true
    items_container.visible = true
    
    var count = items_container.get_child_count()
    if count == 0: return
    
    # 环形均匀分布计算: 弧度 dir = (2 * PI * i) / count
    for i in range(count):
        var item = items_container.get_child(i)
        var dir = (2.0 * PI * i) / count
        var target_pos = Vector2(cos(dir), sin(dir)) * radius
        
        item.update_rotation(dir)
        item.animate_position(target_pos, 0.2)

# 关闭环形选择器
func close() -> void:
    if not is_open: return
    is_open = false
    selected_item = null
    
    # 收回至中心点
    var count = items_container.get_child_count()
    for i in range(count):
        var item = items_container.get_child(i)
        item.animate_position(Vector2.ZERO, 0.15)
    
    # 动画播放完毕后隐藏容器 (Callback 延迟)
    get_tree().create_timer(0.15).timeout.connect(func():
        if not is_open:
            items_container.visible = false
    )

# 动态创建/加装道具
func add_item(status: ItemStatus) -> void:
    var item_inst = wheel_item_scene.instantiate()
    items_container.add_child(item_inst) # 必须先 add_child 确保节点 ready
    item_inst.init_item(status)
# 挂载节点: WheelItem (Area2D - Collision Layer: 8, Mask: 0) | Tween 动画与旋转控制 
extends Area2D
class_name WheelItem

@onready var sprite_2d: Sprite2D = $Sprite2D
@onready var panel: Control = $Panel

var status: ItemStatus
var tween_scale: Tween
var tween_pos: Tween
var tween_rot: Tween

func init_item(new_status: ItemStatus) -> void:
    status = new_status
    if is_node_ready():
        _apply_status()

func _ready() -> void:
    if status:
        _apply_status()

func _apply_status() -> void:
    if status and status.item_icon:
        sprite_2d.texture = status.item_icon

# 缓动平滑缩放
func update_scale(target_scale: float) -> void:
    if tween_scale and tween_scale.is_running():
        tween_scale.kill()
    tween_scale = create_tween()
    tween_scale.tween_property(self, "scale", Vector2.ONE * target_scale, 0.1)

# 缓动调整 Panel 边框旋转角
func update_rotation(target_rot: float) -> void:
    if tween_rot and tween_rot.is_running():
        tween_rot.kill()
    tween_rot = create_tween()
    tween_rot.tween_property(panel, "rotation", target_rot, 0.2)

# 缓动位移展开/收回
func animate_position(target_pos: Vector2, duration: float) -> void:
    if tween_pos and tween_pos.is_running():
        tween_pos.kill()
    tween_pos = create_tween()
    tween_pos.tween_property(self, "position", target_pos, duration)\
        .set_trans(Tween.TRANS_QUAD)\
        .set_ease(Tween.EASE_OUT)
# 自定义 Resource 资源文件: 保存道具名称与图片纹理数据 
extends Resource
class_name ItemStatus

@export var item_name: String = ""
@export var item_icon: Texture2D
# 主测试场景脚本: 监听 Alt 按键按下/释放逻辑与 UI 界面数据刷新 
extends Node2D

@onready var wheel: Node2D = $Wheel
@onready var texture_rect: TextureRect = $UI/TextureRect
@onready var label_name: Label = $UI/LabelName

func _ready() -> void:
    # 从全局配置动态初始化道具
    for status in Global.items:
        wheel.add_item(status)

func _input(event: InputEvent) -> void:
    # 判断 Alt 键操作 (可在项目设置中绑定 ui_alt)
    if event.is_action_pressed("ui_alt"):
        wheel.open()
    elif event.is_action_released("ui_alt"):
        # 松开 Alt 时更新选中的道具
        if wheel.selected_item and wheel.selected_item.status:
            var status = wheel.selected_item.status
            texture_rect.texture = status.item_icon
            label_name.text = status.item_name
        
        wheel.close()
