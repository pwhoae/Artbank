# Godot 4.x UI Tween 动画全面解析脚本
# 适用节点挂载：UI 根节点 Control
extends Control

@onready var texture_rect: TextureRect = $TextureRect
@onready var action_button: Button = $Button
@onready var item_box: VBoxContainer = %ItemBox
@onready var item_template: HBoxContainer = %ItemTemplate

# 保存克隆的模板对象
var template_copy: Control

func _ready() -> void:
	# 1. 关键细节：设置中心缩放点 Pivot Offset 为节点中心 (0.5, 0.5)
	# 避免缩放/旋转时围绕左上角偏移
	texture_rect.pivot_offset = texture_rect.size / 2.0
	action_button.pivot_offset = action_button.size / 2.0
	
	# 2. 连接鼠标移入与移出信号 (Mouse Entered / Exited)
	texture_rect.mouse_entered.connect(_on_texture_rect_mouse_entered)
	texture_rect.mouse_exited.connect(_on_texture_rect_mouse_exited)
	
	action_button.mouse_entered.connect(_on_button_mouse_entered)
	action_button.mouse_exited.connect(_on_button_mouse_exited)
	action_button.pressed.connect(_on_button_pressed)
	
	# 3. 初始化列表模板：深拷贝模板并移除原始 UI 节点
	if item_template:
		template_copy = item_template.duplicate() as Control
		item_template.queue_free()

# ==========================================
# 悬停动画通用函数 (Hover Action)
# ==========================================
func image_hover_action(target: Control, target_scale: float, target_rotation_deg: float) -> void:
	# 创建新的 Tween 句柄
	var tween: Tween = create_tween()
	
	# 设置动画过渡曲线类型 (Cubic) 与缓动模式 (Ease In Out)
	tween.set_trans(Tween.TRANS_CUBIC)
	tween.set_ease(Tween.EASE_IN_OUT)
	
	# 开启并行模式，使得缩放(Scale)与旋转(Rotation)同时播放
	tween.set_parallel(true)
	
	# 绑定缩放属性属性变化 (0.2秒)
	tween.tween_property(target, "scale", Vector2.ONE * target_scale, 0.2)
	
	# 绑定旋转属性 (角度转弧度deg_to_rad)
	var rad_angle: float = deg_to_rad(target_rotation_deg)
	tween.tween_property(target, "rotation", rad_angle, 0.2)

# TextureRect 信号响应
func _on_texture_rect_mouse_entered() -> void:
	image_hover_action(texture_rect, 1.3, 5.0) # 放大 1.3 倍，顺时针旋转 5 度

func _on_texture_rect_mouse_exited() -> void:
	image_hover_action(texture_rect, 1.0, 0.0) # 恢复默认大小与 0 度

# Button 信号响应
func _on_button_mouse_entered() -> void:
	image_hover_action(action_button, 1.1, -3.0)

func _on_button_mouse_exited() -> void:
	image_hover_action(action_button, 1.0, 0.0)

# ==========================================
# 列表平滑生成动画 (Staggered VBox Items)
# ==========================================
func _on_button_pressed() -> void:
	generate_animated_list(5)

func generate_animated_list(count: int) -> void:
	# 清空现有子节点
	for child in item_box.get_children():
		child.queue_free()
		
	if not template_copy:
		return
		
	# 循环生成条目并设置逐条弹出动画
	for i in range(count):
		var new_item: Control = template_copy.duplicate() as Control
		new_item.visible = true
		
		# 设置初始状态：横向/纵向缩放归零，透明度(Alpha)归零
		new_item.scale = Vector2(0.0, 1.0) 
		new_item.modulate.a = 0.0
		new_item.pivot_offset = Vector2(0, new_item.size.y / 2.0)
		
		item_box.add_child(new_item)
		
		# 为新插入的条目创建 Tween 动画
		var tween: Tween = create_tween()
		tween.set_trans(Tween.TRANS_CUBIC)
		tween.set_ease(Tween.EASE_OUT)
		tween.set_parallel(true)
		
		# 展开动画：缩放恢复 Vector2(1, 1)，Alpha 恢复 1.0
		tween.tween_property(new_item, "scale", Vector2.ONE, 0.25)
		tween.tween_property(new_item, "modulate:a", 1.0, 0.25)
		
		# 关键实现：每条生成后等待 0.12 秒，创造阶梯式/错落有致的加载效果
		await get_tree().create_timer(0.12).timeout

Godot 场景节点树层级结构

Control (根节点 - 挂载脚本)
├── TextureRect (Godot 图标 / UI 示例图片)
├── Button (触发加载动画的按钮)
└── VBoxContainer (名称: %ItemBox, 勾选 Unique Name)
└── HBoxContainer (名称: %ItemTemplate, 单元格模板)
├── Label (任务名称)
└── Label (奖励金额)

设置说明：

  • 在右键菜单中将 ItemBox 与 ItemTemplate 设为 Access as Unique Name (%名称)，方便 GDScript 直接通过 %ItemBox 索引。
  • 运行脚本后，代码会自动克隆 ItemTemplate 成为通用模板，并用 queue_free() 销毁原始模板，确保干净的界面逻辑。

踩坑排查与关键配置

1. 中心缩放点 (Pivot Offset) 偏移

默认情况下，Godot UI 节点的锚点在左上角 (0, 0)。若直接缩放旋转，动画会围绕左上角倾斜。必须将 Pivot Offset 设置为节点尺寸的一半（或在属性面板 Layout -> Transform 中设置 Ratio 为 0.5, 0.5）。

2. 容器节点 (Container) 内缩放无效问题

放在 VBoxContainer / HBoxContainer 内部的子节点，其默认 Transform 常会被容器重置。解决方案是使用 scale 与 modulate:a 组合，或对列表子项使用局部偏移。

3. 并行动画 (Parallel) 与 顺序动画

如果不调用 tween.set_parallel(true)，Tween 内部的多个 tween_property() 会按顺序依次播放（例如先完放大，再旋转）。开启并行后才能得到同步顺滑的效果。
