Godot 節點樹結構 (Scene Tree Hierarchy)

HealthBar (Control / ProgressBar - 前景綠條, Z-Index = 1)
├── DamageBar (ProgressBar - 背景緩衝條, Z-Index = -1)
└── Timer (Timer - Wait Time: 0.4s, One Shot: True)

DamageBar 設為 Full Rect：確保大小與父節點 HealthBar 完全一致。

StyleBox 設置：將百分比文字隱藏 (Show Percentage = false)，前景色與背景色設定 Flat StyleBox。

GDScript 實現程式碼 (Godot 4.x)

res://scenes/ui/health_bar.gd 
# res://scenes/ui/health_bar.gd
extends ProgressBar

# 子節點引用
@onready var damage_bar: ProgressBar = $DamageBar
@onready var timer: Timer = $Timer

# 可在 Godot 編輯器調整的參數
@export var damage_delay: float = 0.4
@export var use_smooth_tween: bool = true # 是否啟用平滑縮減效果

# 當前生命值，使用 Setter 自動觸發更新
var health: float = 0.0: set = _set_health

func _ready() -> void:
    # 設定 Timer 屬性
    timer.wait_time = damage_delay
    timer.one_shot = true
    
    # 連接計時器超時訊號
    if not timer.timeout.is_connected(_on_timer_timeout):
        timer.timeout.connect(_on_timer_timeout)

## 初始化生命值 (在角色 _ready 時呼叫)
func init_health(p_health: float) -> void:
    health = p_health
    max_value = health
    value = health
    
    # 同步設定 DamageBar 初始數值
    damage_bar.max_value = health
    damage_bar.value = health

## Setter: 每次 health 改變時會自動呼叫
func _set_health(new_health: float) -> void:
    var prev_health = health
    health = clampf(new_health, 0.0, max_value)
    
    # 1. 頂層綠條立即更新 (表現立即受擊感)
    value = health
    
    # 2. 判斷血量變化狀態
    if health < prev_health:
        # 遭受傷害：啟動/重置延遲計時器 (底層白條暫時不動)
        timer.start()
    else:
        # 治療/回血：底層緩衝條立即補滿並停止計時器
        damage_bar.value = health
        timer.stop()

## Timer 時間到：底層 DamageBar 開始追趕縮減
func _on_timer_timeout() -> void:
    if use_smooth_tween:
        # 方案 A: 使用 Tween 讓底條平滑追上 (視覺反饋極佳)
        var tween = create_tween()
        tween.tween_property(damage_bar, "value", health, 0.25)\
            .set_trans(Tween.TRANS_SINE)\
            .set_ease(Tween.EASE_OUT)
    else:
        # 方案 B: 直接賦值快速縮減
        damage_bar.value = health
# res://scenes/entities/enemy.gd
extends CharacterBody2D

@export var max_health: float = 100.0
var current_health: float

# 頭頂血條節點引用
@onready var health_bar = $HealthBar

func _ready() -> void:
    current_health = max_health
    # 初始化頭頂血條
    if health_bar:
        health_bar.init_health(max_health)

## 受到傷害時呼叫
func take_damage(amount: float) -> void:
    current_health -= amount
    
    # 更新血條 UI (觸發 HealthBar Setter)
    if health_bar:
        health_bar.health = current_health
    
    # 死亡判定
    if current_health <= 0:
        die()

func heal(amount: float) -> void:
    current_health = minf(current_health + amount, max_health)
    if health_bar:
        health_bar.health = current_health

func die() -> void:
    # 執行死亡動畫或銷毀
    queue_free()
# res://scenes/ui/boss_hud.gd
extends CanvasLayer

@onready var boss_health_bar = $MarginContainer/VBoxContainer/HealthBar
@onready var hp_label = $MarginContainer/VBoxContainer/HPLabel

func setup_boss(boss_name: String, max_hp: float) -> void:
    boss_health_bar.init_health(max_hp)
    _update_text(max_hp, max_hp)

func on_boss_health_changed(new_hp: float, max_hp: float) -> void:
    boss_health_bar.health = new_hp
    _update_text(new_hp, max_hp)

func _update_text(hp: float, max_hp: float) -> void:
    if hp_label:
        hp_label.text = "%d / %d" % [int(hp), int(max_hp)]

Godot 屬性檢視器 (Inspector) 設定清單

HealthBar (Root):
  • Show Percentage: False
  • Z-Index: 1 (蓋在白條上方)
  • StyleBox Fill: 藍綠色 (FlatStyle)
DamageBar (Child):
  • Anchors Preset: Full Rect
  • Z-Index: -1 (置於綠條後方)
  • StyleBox Fill: 白色或深紅色
