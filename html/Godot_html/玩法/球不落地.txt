extends CharacterBody2D

var base_speed: float = 450.0
@onready var main_node = get_parent()

func _physics_process(delta: float) -> void:
if not main_node.is_playing:
return

var current_speed = base_speed * main_node.ball_speed_multiplier
var collision = move_and_collide(velocity.normalized() * current_speed * delta)

if collision:
	var collider = collision.get_collider()
	var normal = collision.get_normal()
	
	if collider.name == "Paddle":
		# Calculate bounce angle based on hit position on paddle
		var paddle_x = collider.global_position.x
		var hit_offset = (global_position.x - paddle_x) / 75.0 # Paddle width half is ~75
		var bounce_angle = hit_offset * deg_to_rad(60)
		
		velocity = Vector2.UP.rotated(bounce_angle) * current_speed
		main_node.play_sound(600.0, 0.08)
	else:
		# Wall bounces
		velocity = velocity.bounce(normal)
		main_node.play_sound(400.0, 0.05)
		
# Bottom wall limit (Game Over)
if position.y > 1280:
	main_node.game_over()
