// Godot 4 Control Architecture Node Hierarchy: 
CanvasLayer (EndfieldUI)
  ├── ViewportContainer (GameWorld)
  ├── Control (HUD_Overlay)
  │     ├── MarginContainer -> TopBar
  │     └── MarginContainer -> BottomSkills
  ├── Control (RadialMenuOverlay) - Visible/Hidden toggle
  │     ├── TextureRect (RadarCenter)
  │     └── Control (OrbitButtons)
  └── Control (EventCenter) - UI Panel
        ├── HBoxContainer
        │     ├── VBoxContainer (TabList)
        │     └── PanelContainer (ContentPanel)
// GDScript: UI_Manager.gd (Handling Esc/Tab Menu Transitions & Tweening) 
extends Control

@onready var radial_menu: Control = $RadialMenuOverlay
@onready var event_center: Control = $EventCenter
@onready var hud: Control = $HUD_Overlay

enum UIState { GAMEPLAY, RADIAL_MENU, EVENT_CENTER }
var current_state: UIState = UIState.GAMEPLAY

func _unhandled_input(event: InputEvent) -> void:
    if event.is_action_pressed("ui_focus_next"): # TAB key
        if current_state == UIState.GAMEPLAY:
            switch_state(UIState.RADIAL_MENU)
        elif current_state == UIState.RADIAL_MENU:
            switch_state(UIState.GAMEPLAY)
    elif event.is_action_pressed("ui_cancel"): # ESC key
        switch_state(UIState.GAMEPLAY)

func switch_state(new_state: UIState) -> void:
    current_state = new_state
    
    # Animate visibility with Godot 4 Tween
    var tween = create_tween().set_parallel(true)
    
    match current_state:
        UIState.GAMEPLAY:
            tween.tween_property(radial_menu, "modulate:a", 0.0, 0.2)
            tween.tween_property(event_center, "modulate:a", 0.0, 0.2)
            tween.chain().tween_callback(func(): 
                radial_menu.hide()
                event_center.hide()
            )
        UIState.RADIAL_MENU:
            radial_menu.show()
            radial_menu.modulate.a = 0.0
            tween.tween_property(radial_menu, "modulate:a", 1.0, 0.2)
            tween.tween_property(event_center, "modulate:a", 0.0, 0.2)
        UIState.EVENT_CENTER:
            event_center.show()
            event_center.modulate.a = 0.0
            tween.tween_property(event_center, "modulate:a", 1.0, 0.25)
            tween.tween_property(radial_menu, "modulate:a", 0.0, 0.2)
