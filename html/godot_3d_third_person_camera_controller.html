在遊戲開發中，通常需要按下 Alt 顯示游標來操作裝備欄、地圖或 UI 功能。放開後自動重新捕捉滑鼠繼續旋轉攝影機。

CameraPivot.gd Godot 4.x 推薦使用 (`SpringArm3D` + `Camera3D`) 
# ==============================================================================
# Godot 3D 第三人稱攝影機控制器 (SpringArm3D)
# 挂載對象: 建議掛載於 CharacterBody3D 子節點的 Node3D (Pivot) 或 SpringArm3D 本身
# ==============================================================================
extends Node3D

@export_group("Camera Sensitivity")
## 滑鼠旋轉靈敏度
@export var mouse_sensitivity: float = 0.0025
## 上下仰角最小值 (Degrees)
@export var min_pitch: float = -60.0
## 上下仰角最大值 (Degrees)
@export var max_pitch: float = 75.0
## 反轉 Y 軸
@export var invert_y: bool = false

@export_group("Camera Distance / Zoom")
## 滾輪縮放速度
@export var zoom_speed: float = 0.5
## 最小視角距離
@export var min_zoom: float = 1.5
## 最大視角距離
@export var max_zoom: float = 10.0
## 縮放與旋轉平滑插值速度
@export var smoothness: float = 15.0

@export_group("Mouse Capture Controls")
## 是否啟用 Alt 鍵釋放滑鼠功能
@export var enable_alt_release: bool = true

# 內部狀態變數
var pitch: float = 0.0
var yaw: float = 0.0
var target_zoom: float = 5.0
var is_mouse_captured: bool = true

@onready var spring_arm: SpringArm3D = $SpringArm3D if has_node("SpringArm3D") else null

func _ready() -> void:
	# 遊戲開始時鎖定並隱藏滑鼠游標
	capture_mouse(true)
	
	# 如果有 SpringArm3D，讀取初始長度
	if spring_arm:
		target_zoom = spring_arm.spring_length
		# 避免攝影機被玩家自己的碰撞體擋住 (請依據需求調整類型)
		spring_arm.add_excluded_object(get_parent().get_rid())

func _unhandled_input(event: InputEvent) -> void:
	# 1. 處理 Alt 鍵解鎖 / 恢復滑鼠控制
	if enable_alt_release and event is InputEventKey:
		if event.keycode == KEY_ALT:
			if event.is_pressed() and not event.is_echo():
				# 按下 Alt -> 釋放滑鼠游標
				capture_mouse(false)
			elif event.is_released():
				# 放開 Alt -> 重新捕捉滑鼠
				capture_mouse(true)

	# 2. 如果滑鼠未鎖定，不處理攝影機旋轉
	if not is_mouse_captured:
		# 允許玩家點擊畫面重新鎖定滑鼠
		if event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT and event.is_pressed():
			capture_mouse(true)
		return

	# 3. 滑鼠移動 -> 控制角度 (Pitch & Yaw)
	if event is InputEventMouseMotion:
		var dir_y = -1.0 if invert_y else 1.0
		yaw -= event.relative.x * mouse_sensitivity
		pitch -= event.relative.y * mouse_sensitivity * dir_y
		
		# 限制 Pitch 上下俯仰角度，避免翻轉攝影機
		pitch = clamp(pitch, deg_to_rad(min_pitch), deg_to_rad(max_pitch))

	# 4. 滑鼠滾輪 -> 縮放視角距離 (Zoom)
	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_WHEEL_UP and event.is_pressed():
			target_zoom = max(min_zoom, target_zoom - zoom_speed)
		elif event.button_index == MOUSE_BUTTON_WHEEL_DOWN and event.is_pressed():
			target_zoom = min(max_zoom, target_zoom + zoom_speed)

func _process(delta: float) -> void:
	# 平滑應用旋轉 (Lerp)
	rotation.y = lerp_angle(rotation.y, yaw, delta * smoothness)
	rotation.x = lerp_angle(rotation.x, pitch, delta * smoothness)

	# 平滑縮放 SpringArm3D 彈簧臂長度
	if spring_arm:
		spring_arm.spring_length = lerp(spring_arm.spring_length, target_zoom, delta * smoothness)

## 切換滑鼠鎖定模式
func capture_mouse(captured: bool) -> void:
	is_mouse_captured = captured
	if captured:
		Input.mouse_mode = Input.MOUSE_MODE_CAPTURED
	else:
		Input.mouse_mode = Input.MOUSE_MODE_VISIBLE

Godot 節點結構設置 (Node Hierarchy)

在 Godot 3D 專案中，請依照以下結構層級配置 CharacterBody3D 與攝影機：

🤖 CharacterBody3D (Player角色主體)
├── 🧱 CollisionShape3D
├── 🎨 MeshInstance3D
└── 🎯 CameraPivot (Node3D) ← 挂載本 GDScript
└── 🏹 SpringArm3D
└── 📷 Camera3D

💡 設定重點提示：

  • SpringArm3D 作用：自動檢測地形/牆壁碰撞，當視角後方有牆時會自動縮短距離，防止攝影機穿牆。
  • Camera3D 位置：將 Camera3D 設為 SpringArm3D 的子節點，且將 Camera3D 的 Transform 歸零 (`Position (0, 0, 0)`)。
  • SpringArm 偏移量：可調整 SpringArm3D 的 Position.y (如 1.5m) 作為人頭高度的旋轉中心點。
  • Godot 3.x 語法差異：若使用 Godot 3，請將 `@export` 改為 `export`，`@onready` 改為 `onready`。
