Godot 4 3D 旋转卡片完整实现方案

包含 3D Viewport 贴图与 2D Shader 两种方法，复制即可直接投入生产项目使用。

1 建立 `Node3D` 作为根节点，添加 `MeshInstance3D` (设置为 QuadMesh)。
2 创建 `SubViewport` 节点，里面放入你用 Godot Control (Button/Label) 画好的 2D 卡片。
3 在 Mesh3D 材质的 `Albedo Texture` 中指定 `ViewportTexture` 绑定到 SubViewport！
RotatingCard3D.gd (Godot 4.x) 
# ==============================================================================
# Godot 4.x - 3D 旋转卡片逻辑控制脚本
# 附加节点: Node3D (卡片容器)
# ==============================================================================
extends Node3D

@export_category("3D 旋转参数")
@export var auto_rotate: bool = true          # 是否开启 360° 自转
@export var rotate_speed: float = 1.5         # 旋转速度 (rad/s)
@export var tilt_angle_deg: float = 8.0        # X 轴倾斜角度
@export var enable_hover_pause: bool = true   # 鼠标悬停时暂停自转

@onready var mesh_instance_3d: MeshInstance3D = $MeshInstance3D
@onready var sub_viewport: SubViewport = $SubViewport

var is_hovered: bool = false
var target_rotation_y: float = 0.0

func _ready() -> void:
	# 确保材质设置为双面渲染 (Cull Mode = Disabled)，以便能看到背面
	var mat = mesh_instance_3d.get_active_material(0) as StandardMaterial3D
	if mat:
		mat.cull_mode = BaseMaterial3D.CULL_DISABLED
		# 关联 SubViewport 动态纹理
		var vp_tex = sub_viewport.get_texture()
		mat.albedo_texture = vp_tex

	# 设置初始倾斜角
	rotation_degrees.x = tilt_angle_deg

func _process(delta: float) -> void:
	if auto_rotate and not (enable_hover_pause and is_hovered):
		# 持续 360 度自转
		rotate_y(rotate_speed * delta)

# 响应外部卡片切换或点击翻转命令
func flip_to_back() -> void:
	var tween = create_tween().set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)
	tween.tween_property(self, "rotation_degrees:y", rotation_degrees.y + 180.0, 0.6)

func set_category_index(index: int) -> void:
	# 播放转圈动画并更新 SubViewport 内的 2D 文本
	var tween = create_tween().set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)
	tween.tween_property(self, "rotation_degrees:y", rotation_degrees.y + 360.0, 0.8)
	
	# 更新 2D 界面内容
	if sub_viewport.has_node("PriceCard2D"):
		sub_viewport.get_node("PriceCard2D").update_content(index)
2D Shader 方案优点： 不需要创建 3D 场景与 Viewport，直接在普通 CanvasItem (TextureRect / Control) 上添加 Shader 即可渲染 3D 透视旋转，极大节约 DrawCall 与显存！
Card3DRotation2D.gdshader (Godot 4.x) 
// ==============================================================================
// Godot 4.x CanvasItem Shader: 2D 控件的 3D 透视 360° 旋转
// ==============================================================================
shader_type canvas_item;

uniform float y_rot : hint_range(-180.0, 180.0) = 0.0; // Y 轴旋转角度
uniform float x_rot : hint_range(-90.0, 90.0) = 10.0;   // X 轴倾斜角度
uniform float inset : hint_range(0.0, 1.0) = 0.0;      // 缩放边距
uniform float fov : hint_range(1.0, 179.0) = 90.0;     // 视角场强

// 3D 旋转矩阵计算
mat4 rotation_matrix(vec3 axis, float angle) {
    axis = normalize(axis);
    float s = sin(angle);
    float c = cos(angle);
    float oc = 1.0 - c;
    
    return mat4(
        vec4(oc * axis.x * axis.x + c,           oc * axis.x * axis.y - axis.z * s, oc * axis.z * axis.x + axis.y * s, 0.0),
        vec4(oc * axis.x * axis.y + axis.z * s, oc * axis.y * axis.y + c,           oc * axis.y * axis.z - axis.x * s, 0.0),
        vec4(oc * axis.z * axis.x - axis.y * s, oc * axis.y * axis.z + axis.x * s, oc * axis.z * axis.z + c,           0.0),
        vec4(0.0, 0.0, 0.0, 1.0)
    );
}

void vertex() {
	float rad_y = radians(y_rot);
	float rad_x = radians(x_rot);
	
	// 将 2D 顶点坐标转换为以中心点为原点的 3D 空间坐标
	vec3 p = vec3(VERTEX - (1.0 / TEXTURE_PIXEL_SIZE) * 0.5, 0.0);
	
	mat4 rot = rotation_matrix(vec3(0.0, 1.0, 0.0), rad_y) * rotation_matrix(vec3(1.0, 0.0, 0.0), rad_x);
	p = (rot * vec4(p, 1.0)).xyz;
	
	// 透视投影矩阵收缩计算
	float z_dist = 500.0 / tan(radians(fov * 0.5));
	p.xy *= z_dist / (z_dist - p.z);
	
	VERTEX = p.xy + (1.0 / TEXTURE_PIXEL_SIZE) * 0.5;
}

void fragment() {
	// 背面剔除逻辑，防止镜像文本
	float rad_y = radians(y_rot);
	if (cos(rad_y) < 0.0) {
		// 当旋转到背面时隐藏或替换贴图
		COLOR.a = 0.0;
	} else {
		COLOR = texture(TEXTURE, UV);
	}
}
